\section{Depth scaling of the RF-LR NTK}\label{sec:depth-scaling}

\subsection{Main results}\label{subsec:rflr-terjek-generalization}

\paragraph{Random correlations and deterministic proxy.}
At each layer \( \ell \ge 2 \), the correlation \( \rho_\ell = \cos\angle(\bm{h}^{(\ell-1)}(x), \bm{h}^{(\ell-1)}(x')) \) is a \emph{random variable}: given the population correlation from the previous layer, Fisher's law and Lemma~\ref{lem:gaussian-sample-cosine} give \( \mathbb{E}[(\rho_\ell - \varrho(\rho_{\ell-1}))^2] \le C/r \) and \( \mathbb{P}(|\rho_\ell - \varrho(\rho_{\ell-1})| \ge t) \le 6\exp(-c\,r\,t^2) \), so fluctuations are \( O(1/\sqrt{r}) \) sub-Gaussian. The NTK recursion passes through this chain of random correlations. The \emph{deterministic proxy} replaces \( \rho_\ell \) by the deterministic iterates \( \rho_k = \varrho^{\circ(k-1)}(\rho_1) \) and analyzes the scalar recursion~\eqref{eq:rflr-scalar-mean-ntk-rec}; for large \(r\) the random kernel concentrates around the proxy (Appendix~\ref{appendix:random-correlation-chain}); a formal bound \(\|\hat{K}-K_{\mathrm{proxy}}\|_{\mathrm{op}}=o_P(1)\) as \(r\to\infty\) and its implication for empirical conditioning are given in Appendix~\ref{appendix:proxy-empirical-concentration}. In the equicorrelated case, a full proof yields \(\|(\hat{K}-K_{\mathrm{proxy}})|_{\mathbf{1}^\perp}\|_{\mathrm{op}}=O_P(L/r+1/\sqrt{r})\) (Theorem~\ref{thm:equicorrelated-op-bound}). All condition-number statements in this section refer to the \emph{proxy} Gram matrix; we give a lower bound on its condition number and, in one special case, exact conditioning with high probability.

\paragraph{Depth--rank scaling.}
The recursion contains an explicit \(1/r\) bottleneck factor, which leads to saturation of the kernel magnitude as depth grows, while the diagonal--off-diagonal gap decays with depth in the proxy recursion (the same \(1-\rho_k=\Theta(k^{-2})\) correlation alignment as for full-width MLPs at the edge of chaos~\cite{terjek2025mlpsateoc,pmlr-v163-hayou22a}). Theorem~\ref{thm:rflr-depth-scaling-mean} analyzes this deterministic proxy, which approximates the mean path of the random recursion for large \(r\). We refer to Appendix~\ref{appendix:mean-depth-theorem-preliminaries} for notation and kernel expansions.

\begin{theorem}[Depth scaling for the deterministic proxy RF-LR NTK]\label{thm:rflr-depth-scaling-mean}
Fix \(r>1\) and \(\rho_1\in(-1,1)\). Let \(\Theta^{(k)}(\rho_1)\) be the deterministic proxy defined by~\eqref{eq:rflr-scalar-mean-ntk-rec}, with \(\rho_k = \varrho^{\circ(k-1)}(\rho_1)\) deterministic. Then (as for MLPs at EOC~\cite{terjek2025mlpsateoc,pmlr-v163-hayou22a}):
\begin{itemize}
  \item (\textbf{Correlation alignment}) The correlation recursion aligns polynomially:
  \[
  1-\rho_k \;=\; \Theta(k^{-2}).
  \]
  \item (\textbf{Kernel saturation}) \(\Theta^{(k)}(\rho_1)\to \Theta_\star(r)\) as \(k\to\infty\), where \(\Theta_\star(r)\) is the fixed point of the limiting recursion obtained by setting \(\dot{s}=s=1/2\), namely
  \[
  \Theta_\star(r) \;=\; \frac{2r+1}{2r-1}.
  \]
  Moreover,
  \[
  \Theta^{(k)}(\rho_1)-\Theta_\star(r) \;=\; O(k^{-1}).
  \]
  \item (\textbf{Depth-induced gap decay}) Let \(\Theta^{(k)}_{\mathrm{diag}}:=\Theta^{(k)}(1)\) (on-diagonal value, i.e. identical inputs) and \(\Theta^{(k)}_{\mathrm{off}}:=\Theta^{(k)}(\rho_1)\) (off-diagonal value for an input pair with initial cosine similarity \(\rho_1\)). Then
  \[
  \Theta^{(k)}_{\mathrm{diag}}-\Theta^{(k)}_{\mathrm{off}} \;=\; \Theta\!\Big(\frac{1}{r\,k}\Big),
  \]
  for large \(k\).
\end{itemize}
\end{theorem}

\begin{proof}
See Appendix~\ref{appendix:proof-rflr-depth-scaling-mean}.
\end{proof}

\begin{proposition}[Lower bound on condition number for the proxy]\label{prop:condition-number-depth}
In the setting of Theorem~\ref{thm:rflr-depth-scaling-mean}, consider the deterministic proxy kernel \(\Theta^{(L)}(\rho)\) and form the \(n\times n\) Gram matrix \(\mathbf{M}\) with entries \(M_{ij}=\Theta^{(L)}(\rho_{ij})\) for pairwise input cosine similarities \(\rho_{ij}=\langle x_i,x_j\rangle/(\|x_i\|\|x_j\|)\in(-1,1)\). Let \(\lambda_{\max}\) and \(\lambda_{\min}\) denote the maximum and minimum eigenvalues of \(\mathbf{M}\) restricted to \(\mathbf{1}^\perp\). Assume: (i)~\(n\ge 3\); (ii)~there exists a pair \((i,j)\) with \(\rho_{ij}\) bounded away from \(1\); (iii)~the dataset is not equicorrelated (the \(\rho_{ij}\) for \(i\neq j\) are not all equal). Then \(\lambda_{\max}=\Theta(1)\), \(\lambda_{\min}\le O(1/(rL))\), and the condition number \(\kappa=\lambda_{\max}/\lambda_{\min}\) satisfies \(\kappa\ge\Omega(r\cdot L)\). When assumption~(iii) is violated (equicorrelated data), the proposition does not apply; instead Corollary~\ref{cor:conditioning-equicorrelated-highdim} gives the best-case \(\kappa_\perp=1\) (and similarly for approximately equicorrelated high-dimensional data).
\end{proposition}
\begin{proof}
See Appendix~\ref{appendix:proof-condition-number}.
\end{proof}

\begin{theorem}[Operator norm on \(\mathbf{1}^\perp\) in the equicorrelated case]\label{thm:equicorrelated-op-bound}
Fix \(n,L\ge 1\), \(r\ge 2\), and the equicorrelated setup \(\rho_{1,ii}=1\), \(\rho_{1,ij}=\rho_0\) for \(i\neq j\) with \(\rho_0\in(-1,1)\). Let \(\hat{K}\) be the \(n\times n\) empirical RF-LR NTK Gram matrix and \(K_{\mathrm{proxy}}\) the deterministic proxy with entries \((K_{\mathrm{proxy}})_{ij}=\Theta^{(L)}(\rho_{1,ij})\) (Definition~\ref{def:deterministic-proxy}). Then there exist constants \(C,c>0\) (depending on \(L,\rho_0\)) such that for any \(\epsilon>0\),
\[
\mathbb{P}\!\left(\bigl\|\bigl(\hat{K}-K_{\mathrm{proxy}}\bigr)\big|_{\mathbf{1}^\perp}\bigr\|_{\mathrm{op}}\;\ge\; \epsilon\right)
\;\le\; C\,\exp\!\left(-c\,\frac{r\,\epsilon^2}{L^2}\right)\;+\; C\,\exp\!\left(-c\,\frac{r\,\epsilon}{L}\right).
\]
In particular, \(\bigl\|\bigl(\hat{K}-K_{\mathrm{proxy}}\bigr)\big|_{\mathbf{1}^\perp}\bigr\|_{\mathrm{op}}=O_P(L/r+1/\sqrt{r})\) as \(r\to\infty\) with \(n,L\) fixed. Hence if \(r=O(L^2)\), then \(\bigl\|\bigl(\hat{K}-K_{\mathrm{proxy}}\bigr)\big|_{\mathbf{1}^\perp}\bigr\|_{\mathrm{op}}=O_P(L/r)\).
\end{theorem}
\begin{proof}
See Appendix~\ref{appendix:proxy-empirical-concentration}.
\end{proof}

\begin{corollary}[Exact conditioning for equicorrelated and high-dimensional random data]\label{cor:conditioning-equicorrelated-highdim}
In the setting of Proposition~\ref{prop:condition-number-depth}, let \(\kappa_\perp=\lambda_{\max}(\mathbf{M}|_{\mathbf{1}^\perp})/\lambda_{\min}(\mathbf{M}|_{\mathbf{1}^\perp})\) denote the condition number on the mean-zero subspace. The following hold.
\begin{itemize}
  \item \textbf{Equicorrelated data.} If \(\rho_{ij}=\rho_0\) for all \(i\neq j\) (and \(\rho_{ii}=1\)), then on \(\mathbf{1}^\perp\) all \(n-1\) eigenvalues of \(\mathbf{M}\) equal \(\lambda_\perp=\Theta^{(L)}(1)-\Theta^{(L)}(\rho_0)=\Theta(1/(rL))\). Hence
  \[
  \kappa_\perp \;=\; \frac{\lambda_{\max}}{\lambda_{\min}} \;=\; \frac{\lambda_\perp}{\lambda_\perp} \;=\; 1.
  \]
  \item \textbf{High-dimensional random data.} If the inputs \(x_1,\ldots,x_n\in\mathbb{S}^{d-1}\) are drawn i.i.d.\ uniformly from the unit sphere, then as \(d\to\infty\) (with \(n,r,L\) fixed) with high probability \(\max_{i\neq j}|\rho_{ij}|=O(1/\sqrt{d})\), so the proxy Gram matrix is approximately equicorrelated with \(\rho_0=O(1/\sqrt{d})\). On that event, all eigenvalues on \(\mathbf{1}^\perp\) are \(\lambda_\perp=\Theta^{(L)}(1)-\Theta^{(L)}(\bar{\rho})+o(1)=\Theta(1/(rL))(1+o(1))\) with \(\bar{\rho}=O(1/\sqrt{d})\), and they differ from each other by \(o(\lambda_\perp)\). Hence
  \[
  \kappa_\perp \;=\; \frac{\lambda_{\max}}{\lambda_{\min}} \;=\; 1+o(1)\qquad\text{as }d\to\infty.
  \]
  On that high-probability event, the empirical Gram matrix \(\hat{K}\) concentrates around the proxy: \(\|(\hat{K}-K_{\mathrm{proxy}})|_{\mathbf{1}^\perp}\|_{\mathrm{op}}=O_P(L/r+1/\sqrt{r})\) (Theorem~\ref{thm:equicorrelated-op-bound} with \(\rho_0=O(1/\sqrt{d})\)).
\end{itemize}
\end{corollary}
\begin{proof}
See Appendix~\ref{appendix:proof-conditioning-equicorrelated-highdim}.
\end{proof}

\paragraph{Limited scope of exact conditioning.}
Equicorrelated and high-dimensional i.i.d.\ spherical data are special; the general conditioning guarantee in this section is the proxy lower bound (Proposition~\ref{prop:condition-number-depth}). Summary of condition-number scaling:
\begin{itemize}
  \item \textbf{Equicorrelated:} \(\kappa_\perp=1\) (exact; all eigenvalues on \(\mathbf{1}^\perp\) equal \(\lambda_\perp=\Theta(1/(rL))\)).
  \item \textbf{High-dim i.i.d.\ spherical:} \(\kappa_\perp=1+o(1)\) as \(d\to\infty\) (approximately equicorrelated).
  \item \textbf{General (non-equicorrelated):} \(\kappa\ge\Omega(r\cdot L)\); explicit: \(\lambda_{\max}=\Theta(1)\), \(\lambda_{\min}\le O(1/(rL))\), hence \(\kappa=\lambda_{\max}/\lambda_{\min}\ge\Omega(r\cdot L)\).
\end{itemize}

\paragraph{Proxy vs.\ optimization.}\label{rmk:depth-conditioning-message}
Proposition~\ref{prop:condition-number-depth} gives a \emph{lower bound} on the condition number of the \emph{proxy} Gram matrix; we see strong experimental agreement with our theory and proxy predictions, and exact RKHS equivalence holds for three layers, with extension to depth \(L\ge 4\) left to future work. For the proxy and non-equicorrelated data, depth drives the kernel toward near-constant entries (gap \(\Theta(1/(rL))\), as for MLPs at EOC) and \(\kappa\ge\Omega(rL)\). The actual network uses random correlations with \(O(1/\sqrt{r})\) sub-Gaussian fluctuations (Appendix~\ref{appendix:random-correlation-chain}), so for large \(r\) the random kernel concentrates around the proxy. The condition number is relevant for kernel regression (stability and convergence speed depend on \(\lambda_{\max}/\lambda_{\min}\)); under a fixed parameter budget \(O(NLr)\), depth and rank trade off from a conditioning perspective, and the choice of \(r\) and \(L\) is a matter of approximation power (effective depth and RKHS expressivity).
