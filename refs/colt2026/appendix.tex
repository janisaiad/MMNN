\section{Summary of main results}\label{appendix:summary}

This work develops an NTK analysis for low-rank random-feature architectures (RF-LR/MMNN) under explicit assumptions. Appendix~\ref{sec:experiments} reports numerical illustrations that accompany the theory.

\textbf{Main: depth/rank scaling and condition number.} The trainable parameter budget scales as \(O(NLr)\). In the deterministic proxy mean recursion, depth drives correlations toward \(1\) with \(1-\rho_k=\Theta(k^{-2})\) (the same correlation alignment as for full-width MLPs at the edge of chaos~\cite{terjek2025mlpsateoc,pmlr-v163-hayou22a}), the kernel magnitude saturates, and the diagonal--off-diagonal gap decays as \(\asymp 1/(rk)\) (Theorem~\ref{thm:rflr-depth-scaling-mean}). We prove a lower bound on the condition number of the \emph{proxy} Gram matrix, \(\kappa\ge\Omega(r\cdot L)\) (Proposition~\ref{prop:condition-number-depth}), and exact conditioning \(\kappa_\perp=1\) or \(\kappa_\perp=1+o(1)\) for equicorrelated or high-dimensional random data (Corollary~\ref{cor:conditioning-equicorrelated-highdim}). Under a fixed budget \(O(NLr)\), depth and rank trade off from a conditioning perspective; the choice of \(r\) and \(L\) is a matter of approximation power (e.g.\ effective depth and RKHS expressivity).

\textbf{Recursion.} We provide an explicit NTK recursion for arbitrary depth in RF-LR and a closed-form expansion (Theorem~\ref{thm:ntk_recursion}, Corollary~\ref{cor:general_L_layer_ntk}).

\textbf{Low rank is enough for the NTK RKHS (three layers).} In the NTK regime, the three-layer RF-LR mean kernel induces the same RKHS as the shallow ReLU kernel (Corollary~\ref{thm:no_rkhs_advantage}), showing that the bottleneck does not reduce the kernel-regime function class while lowering the trainable parameter scaling from dense \(O(LN^2)\) to \(O(LrN)\).

\textbf{Concentration in the bottleneck dimension.} For the three-layer case, the empirical kernel concentrates around its deterministic limit \(K_\infty(\rho)\) with tails controlled by the bottleneck dimension \(r\) (Theorem~\ref{thm:rkhs_rank_concentration}, Corollary~\ref{cor:three_layer_ntk_concentration}). For equicorrelated data, a full proof in the appendix gives \(\|(\hat{K}-K_{\mathrm{proxy}})|_{\mathbf{1}^\perp}\|_{\mathrm{op}}=O_P(L/r+1/\sqrt{r})\) (Theorem~\ref{thm:equicorrelated-op-bound}), linking proxy and empirical conditioning on the mean-zero subspace.

\section{Notation and standing assumptions}\label{appendix:notation-assumptions}
\subsection{Standing assumptions (expanded)}\label{appendix:assumptions-expanded}
We summarize the conventions used throughout the main text.
\begin{itemize}
    \item \textbf{Sequential infinite width.} Widths \(\bm{n}_1,\ldots,\bm{n}_L\) tend to infinity sequentially under NTK scaling \(1/\sqrt{\bm{n}_\ell}\). Random features \(w^{(\ell)}\) are i.i.d.\ Gaussian and frozen after initialization.
    \item \textbf{Training policy.} Only the readouts \(A^{(\ell)}\) (and layer biases \(c^{(\ell)}\)) are trained; all other parameters are frozen.
    \item \textbf{Bottleneck regime.} For \(\ell\ge 2\), the representation dimension is \(\bm{d}_\ell=r\ll \bm{n}_\ell\). This induces an explicit \(1/r\) factor in the RF-LR NTK recursion at bottleneck layers.
    \item \textbf{Zonal kernels.} By rotational invariance of Gaussian features, base and derivative kernels depend on inputs only through the cosine similarity \(\rho=\langle x,x'\rangle/(\|x\|\,\|x'\|)\). For RKHS results we typically take \(x\in\mathbb{S}^{d-1}\).
    \item \textbf{Homogeneity.} We use Assumption~\ref{assump:homogeneous_activation} (positive 1-homogeneity) for Fisher--Kibble decoupling and for separating radial and angular components in three-layer calculations.
\end{itemize}

\subsection{Finite-width empirical NTK and limiting gradient flow}\label{appendix:empirical-ntk-gradient-flow}
For a finite-width network initialized at random parameters \(\theta_0\), the empirical NTK \(\Theta^{(L)}_{\theta_0}\) is a random kernel (the Gram operator of gradients \(\langle \nabla_\theta f(x;\theta_0),\nabla_\theta f(x';\theta_0)\rangle\)). The kernel gradient flow trajectory \(t\mapsto f_t\) is therefore a random object (measurable with respect to \(\theta_0\)) and satisfies
\[
\frac{d}{dt}f_t(x) \;=\; -\int \Theta^{(L)}_{\theta_0}(x,x')\,\bigl(f_t(x')-y(x')\bigr)\,d\mu(x'),
\]
where \(\mu\) is the data distribution and \(y\) is the target function. In the sequential infinite-width NTK limit, \(\Theta^{(L)}_{\theta_0}\) concentrates and converges in probability to the deterministic kernel \(\Theta^{(L)}\), yielding the corresponding deterministic limiting gradient flow driven by \(\Theta^{(L)}\)~\cite{jacot2018neural}.

\begin{assumption}[Homogeneous activation]\label{assump:homogeneous_activation}
The activation function \( \sigma \) is positively 1-homogeneous:
\[
\sigma(\alpha u) \,=\, \alpha\,\sigma(u)\quad\text{for all } \alpha\ge 0,\; u\in\mathbb{R}.
\]
In particular, \(\ReLU\) satisfies this property. This assumption is used for Fisher--Kibble decoupling (Lemma~\ref{lem:fisher-kibble}) and for separating radial and angular fluctuations in the three-layer kernel.
\end{assumption}

\subsection{\texorpdfstring{EOC scaling and the origin of \(1/r\)}{EOC scaling and the origin of 1/r}}\label{appendix:eoc-scaling}
Let \(\Sigma_w=\mathrm{Cov}(w^{(\ell)}_j)\). Writing \(q^{(\ell)}=\mathbb{E}\|h^{(\ell)}(x)\|^2\) for a fixed unit-norm input, weight independence yields the variance recursion
\[
q^{(\ell)}=\frac{\sigma_A^2}{2}\,q^{(\ell-1)}\,\mathrm{Tr}(\Sigma_w),
\]
The edge-of-chaos (EOC) condition chooses \(\sigma_A\) so that the linearized coefficient equals \(1\),
\[
\frac{\sigma_A^2}{2}\,\mathrm{Tr}(\Sigma_w)=1,
\]
preventing exploding/vanishing signal propagation across depth. In particular, under \(\Sigma_w=I_{\bm{d}_{\ell-1}}/\bm{d}_{\ell-1}\) one has \(\mathrm{Tr}(\Sigma_w)=1\) and \(\sigma_A=\sqrt{2}\).

In the bottleneck regime \(\bm{d}_{\ell-1}=r\) for \(\ell\ge 2\), EOC prescribes \(\Sigma_w=I_r/r\). For ReLU, the (normalized) derivative kernel has the closed form
\[
\bar{\dot{\Sigma}}^{(\ell)}(\rho)=\mathbb{P}(Z>0,Z'>0)=\frac{1}{2}-\frac{\arccos(\rho)}{2\pi}\in\Big[0,\frac12\Big],
\]
so \(\dot{\Sigma}^{(\ell)}(x,x')=\bar{\dot{\Sigma}}^{(\ell)}(\rho_\ell)\) remains \(O(1)\) under EOC and is uniformly bounded by \(1/2\).
The RF-LR NTK recursion (Theorem~\ref{thm:ntk_recursion}) carries an explicit prefactor \(1/r\) at bottleneck layers because gradients propagate through an \(r\)-dimensional bottleneck and the corresponding Jacobian metric concentrates on an \(r\)-dimensional subspace; see Appendix~\ref{appendix:proof-ntk-recursion} for the detailed derivation.

\subsection{Comparison with the classical fully-trained NTK recursion}\label{appendix:classical-ntk-comparison}
For comparison, consider a standard fully-connected MLP where \emph{all} weights (and biases) are trained under NTK parameterization. In the sequential infinite-width limit, the classical NTK recursion takes the form~\cite{jacot2018neural}
\begin{equation}
\label{eq:classical_ntk_recursion}
\Theta_{\mathrm{MLP}}^{(0)}(x,x')=0,\qquad
\Theta_{\mathrm{MLP}}^{(\ell)}(x,x')=\Theta_{\mathrm{MLP}}^{(\ell-1)}(x,x')\,\dot{K}^{(\ell)}(x,x')+K^{(\ell)}(x,x'),\quad \ell\ge 1,
\end{equation}
where \(K^{(\ell)}\) is the (deterministic) NNGP kernel at layer \(\ell\) and \(\dot{K}^{(\ell)}\) the associated derivative kernel (both obtained from the signal-propagation recursion).
Unrolling \eqref{eq:classical_ntk_recursion} yields one contribution per layer (a sum of \(L\) terms, each propagated through subsequent \(\dot{K}^{(k)}\) factors).

In contrast, the RF-LR recursion of Theorem~\ref{thm:ntk_recursion} (i) carries explicit \(1/r\) prefactors at bottleneck layers, and (ii) uses conditional base/derivative kernels \(\Sigma^{(\ell)},\dot{\Sigma}^{(\ell)}\) rather than the classical NNGP kernels \(K^{(\ell)},\dot{K}^{(\ell)}\). These differences lead to the \(2L\)-term expansion in Corollary~\ref{cor:general_L_layer_ntk}, reflecting separate bias-path and fresh-basis contributions, each propagated through subsequent derivative kernels and weighted by bottleneck factors.

\paragraph{Relation to fully trained MLPs at the edge of chaos.}
For fully trained (full-width) MLPs at the edge of chaos, the limiting NTK admits a single-layer sum structure (one contribution per layer) without RF-LR bottleneck prefactors; see Proposition~\ref{prop:mlp-eoc-ntk-formula} for a reference closed form~\cite{terjek2025mlpsateoc}. In contrast, the RF-LR expansion \eqref{eq:ntk_explicit_form} contains \(2L\) terms because the recursion has both bias-path and fresh-basis-path contributions, each propagated through subsequent derivative kernels and weighted by explicit \(1/r\) factors at bottlenecks.

\subsection{Probabilistic recursion, GP concatenation, and bottleneck scaling}\label{appendix:probabilistic-recursion-details}
\paragraph{Random correlation chain and Fisher's law.}\label{appendix:random-correlation-chain}
At each layer \( \ell \ge 2 \), the pair \( (\bm{h}^{(\ell-1)}(x), \bm{h}^{(\ell-1)}(x')) \) is a \(2r\)-dimensional Gaussian conditional on lower-layer randomness. The \emph{layer-\(\ell\) sample correlation}
\[
\rho_\ell \;=\; \cos\angle\bigl(\bm{h}^{(\ell-1)}(x), \bm{h}^{(\ell-1)}(x')\bigr) \;=\; \frac{\langle \bm{h}^{(\ell-1)}(x), \bm{h}^{(\ell-1)}(x') \rangle}{\|\bm{h}^{(\ell-1)}(x)\|\,\|\bm{h}^{(\ell-1)}(x')\|}
\]
is a \emph{random variable}. Given the population correlation \( \rho = \varrho(\rho_{\ell-1}) \) (the conditional mean under the ReLU cosine map), Fisher's law~\cite{fisher1915probable,hotelling1953new} and Lemma~\ref{lem:gaussian-sample-cosine} give \( \mathbb{E}[(\rho_\ell - \rho)^2] \le C/r \) and \( \mathbb{P}(|\rho_\ell - \rho| \ge t) \le 6\exp(-c\,r\,t^2) \). Thus the chain \( \rho_1 \to \rho_2 \to \cdots \to \rho_L \) is a \emph{Markov chain of random variables} with \( O(1/\sqrt{r}) \) fluctuations at each transition. The NTK recursion passes through these random correlations successively: \( \Sigma^{(\ell)} = \bar{\Sigma}^{(\ell)}(\rho_\ell) \), \( \dot{\Sigma}^{(\ell)} = \bar{\dot{\Sigma}}^{(\ell)}(\rho_\ell) \) with random \( \rho_\ell \). The deterministic proxy (Definition~\ref{def:deterministic-proxy}) replaces this random path by the deterministic iterates \( \rho_k = \varrho^{\circ(k-1)}(\rho_1) \), which represent the mean/infinite-\(r\) limit path.

\paragraph{Kernel randomness and GP concatenation.}
For \( \ell > 1 \), \( \Sigma^{(\ell)} \) and \( \dot{\Sigma}^{(\ell)} \) are random fields because they depend on \( \bm{h}^{(\ell-1)} \), which is the output of a (conditional) Gaussian process. By Theorem~\ref{thm:gp-composition}, the hidden states form a composition of GPs: each layer's output is conditionally Gaussian given the previous layer, but the unconditional law is a deep GP. In the bottleneck regime, the pair \( (\bm{h}^{(\ell-1)}(x), \bm{h}^{(\ell-1)}(x')) \) is a \(2r\)-dimensional Gaussian conditional on lower-layer randomness, with covariance determined by \( \Sigma^{(\ell-1)} \). The base kernel \( \Sigma^{(\ell)} \) and derivative kernel \( \dot{\Sigma}^{(\ell)} \) are thus expectations over the frozen features \( \bm{w}^{(\ell)} \), conditional on this Gaussian pair, and inherit randomness through \( \Sigma^{(\ell-1)} \).

\paragraph{Bottleneck scaling and explicit ReLU bounds.}
At a bottleneck layer \( \ell \ge 2 \), the RF-LR NTK recursion carries an explicit prefactor \(1/r\) (Theorem~\ref{thm:ntk_recursion}). For ReLU and isotropic weights with \( \mathrm{Cov}(w)=I_r/r \), one has
\[
\dot{\Sigma}^{(\ell)}(x,x') \;=\; \mathbb{E}_w\!\Big[\dot{\sigma}\!\big(w^\top \bm{h}^{(\ell-1)}(x)\big)\,\dot{\sigma}\!\big(w^\top \bm{h}^{(\ell-1)}(x')\big)\,\|w\|^2\Big] \;=\; \bar{\dot{\Sigma}}^{(\ell)}(\rho_\ell),
\]
where \( \rho_\ell = \cos\angle(\bm{h}^{(\ell-1)}(x), \bm{h}^{(\ell-1)}(x')) \) and \( \bar{\dot{\Sigma}}^{(\ell)}(\rho) \) denotes the corresponding scalar derivative kernel evaluated at cosine similarity \( \rho\in[-1,1] \).
Under the EOC normalization \( \mathrm{Tr}(\mathrm{Cov}(w))=1 \), the ReLU derivative kernel has the explicit form
\[
\bar{\dot{\Sigma}}^{(\ell)}(\rho)=\mathbb{P}(Z>0,Z'>0)=\frac{1}{2}-\frac{\arccos(\rho)}{2\pi}\in\Big[0,\frac12\Big],
\]
where \((Z,Z')\) is centered Gaussian with correlation \(\rho\). In particular, \(\sup_{\rho\in[-1,1]}\bar{\dot{\Sigma}}^{(\ell)}(\rho)=1/2\), uniformly in \(\ell\).
Similarly, \( \Sigma^{(\ell)}(x,x')=\mathbb{E}_w[\sigma(w^\top \bm{h}^{(\ell-1)}(x))\,\sigma(w^\top \bm{h}^{(\ell-1)}(x'))] \) is an \(O(1)\) scalar kernel, and the \(1/r\) factors enter through the recursion.

\paragraph{Probabilistic recursion (nested expectations).}
The mean (probabilistic) NTK \( \tilde{\Kop}^{(L)}=\mathbb{E}[\Kop^{(L)}] \) satisfies a recursion involving expectations of products:
\begin{equation}
\label{eq:mean_ntk_recursion}
\tilde{\Kop}^{(1)} \;=\; 1 \;+\; \mathbb{E}\!\big[\Sigma^{(1)}\big],\qquad
\tilde{\Kop}^{(\ell)} \;=\; 1 \;+\; \frac{1}{r}\,\mathbb{E}\!\big[\Kop^{(\ell-1)}\cdot\dot{\Sigma}^{(\ell)}\big] \;+\; \frac{1}{r}\,\mathbb{E}\!\big[\Sigma^{(\ell)}\big],\quad \ell \ge 2.
\end{equation}
To emphasize the nested structure, write \(\mathbb{E}_{\le \ell-1}\) for expectation over all randomness up to depth \(\ell-1\), and \(\mathbb{E}_{w^{(\ell)}}[\cdot\,|\,\bm{h}^{(\ell-1)}]\) for expectation over the frozen feature direction \(w^{(\ell)}\) at layer \(\ell\), conditional on the previous hidden states. Then the mean kernels are iterated conditional expectations:
\[
\mathbb{E}\!\big[\Sigma^{(\ell)}(x,x')\big]
\;=\;
\mathbb{E}_{\le \ell-1}\Big[
\mathbb{E}_{w^{(\ell)}}\big[\sigma(w^{\top}\bm{h}^{(\ell-1)}(x))\,\sigma(w^{\top}\bm{h}^{(\ell-1)}(x'))\;\big|\;\bm{h}^{(\ell-1)}\big]
\Big],
\]
\[
\mathbb{E}\!\big[\Kop^{(\ell-1)}(x,x')\,\dot{\Sigma}^{(\ell)}(x,x')\big]
\;=\;
\mathbb{E}_{\le \ell-1}\Big[
\Kop^{(\ell-1)}(x,x')\cdot
\mathbb{E}_{w^{(\ell)}}\big[\dot{\sigma}(w^{\top}\bm{h}^{(\ell-1)}(x))\,\dot{\sigma}(w^{\top}\bm{h}^{(\ell-1)}(x'))\,\|w\|^2\;\big|\;\bm{h}^{(\ell-1)}\big]
\Big].
\]
Since \( \Kop^{(\ell-1)} \) and \( \Sigma^{(\ell)},\dot{\Sigma}^{(\ell)} \) share the randomness of \( \bm{h}^{(\ell-1)} \), one has \( \mathbb{E}[\Kop^{(\ell-1)}\cdot\dot{\Sigma}^{(\ell)}] \neq \tilde{\Kop}^{(\ell-1)}\cdot\mathbb{E}[\dot{\Sigma}^{(\ell)}] \) in general; controlling this dependence (and the resulting nested conditional expectations) is the main technical point for a rigorous deep mean-kernel theory.

\paragraph{Exponential depth suppression (ReLU, bottlenecks).}
Assume for simplicity that all layers \(\ell\ge 2\) are bottlenecks with the same rank \(r\). For ReLU under EOC we have the explicit uniform bound
\[
\bar{\dot{\Sigma}}^{(\ell)}(\rho)\le c_0:=\frac12,\qquad \forall \rho\in[-1,1],\ \forall \ell\ge 2.
\]
From the closed form \eqref{eq:ntk_explicit_form}, the contribution from layer \(\ell\) to \(\Kop^{(L)}\) is multiplied by \(\prod_{k=\ell+1}^{L}\dot{\Sigma}^{(k)}\). Writing \(\dot{\Sigma}^{(k)}=\bar{\dot{\Sigma}}^{(k)}(\rho_k)\), we obtain
\[
\frac{1}{r^{L-\ell}}\prod_{k=\ell+1}^{L}\dot{\Sigma}^{(k)}
\;\le\;
\frac{1}{r^{L-\ell}}\,\prod_{k=\ell+1}^{L}\bar{\dot{\Sigma}}^{(k)}(\rho_k)
\;\le\;\frac{c_0^{L-\ell}}{r^{L-\ell}}.
\]
Hence, contributions from early layers (\(\ell\ll L\)) decay exponentially in \(L-\ell\) when \(c_0/r<1\), and the NTK becomes increasingly localized to the top of the network as depth grows.

\begin{proposition}[Effective depth decomposition and a \(\log r\) window]\label{prop:depth_decay}
Assume the uniform derivative-kernel bound \(\dot{\Sigma}^{(\ell)}(x,x')\le c_0\) for all \(\ell\ge 2\) and all input pairs \((x,x')\), and assume that for a given \((x,x')\) one has a uniform base-kernel bound
\[
|\Sigma^{(\ell)}(x,x')|\le C_\Sigma(x,x'),\qquad \forall \ell\in\{1,\dots,L\}.
\]
Fix \(L\ge 2\), \(r>c_0\), and an integer \(m\in\{1,\dots,L\}\). Define the \emph{top-\(m\) truncation} \(\Kop^{(L)}_{\mathrm{top},m}(x,x')\) by restricting both sums in the explicit expansion \eqref{eq:ntk_explicit_form} to indices \(\ell\in\{L-m+1,\dots,L\}\). Then
\[
\big|\Kop^{(L)}(x,x')-\Kop^{(L)}_{\mathrm{top},m}(x,x')\big|
\;\le\;
\frac{1+C_\Sigma(x,x')}{1-c_0/r}\left(\frac{c_0}{r}\right)^m.
\]
In particular, taking \(m=\lceil \alpha \log r\rceil\) (for any \(\alpha>0\)) yields a \(\log r\) effective-depth window: the contribution of layers \(\ell\le L-\lceil \alpha \log r\rceil\) is at most
\[
\big|\Kop^{(L)}(x,x')-\Kop^{(L)}_{\mathrm{top},\lceil \alpha \log r\rceil}(x,x')\big|
\;\le\;
\frac{1+C_\Sigma(x,x')}{1-c_0/r}\left(\frac{c_0}{r}\right)^{\lceil \alpha \log r\rceil}
\;=\;o(r^{-p})
\]
for every fixed \(p>0\) as \(r\to\infty\).
\end{proposition}
\paragraph{Proof sketch.} The remainder is the sum of terms in \eqref{eq:ntk_explicit_form} with \(\ell\le L-m\); each such term carries a factor \(\prod_{k=\ell+1}^{L}\dot{\Sigma}^{(k)}\le c_0^{L-\ell}\) and hence \((c_0/r)^{L-\ell}\). Summing over \(\ell\le L-m\) yields a geometric-series tail; substituting \(m=\lceil\alpha\log r\rceil\) gives the stated decay.
\begin{proof}
By definition of \(\Kop^{(L)}_{\mathrm{top},m}\), the remainder is the sum of the terms in \eqref{eq:ntk_explicit_form} with \(\ell\le L-m\). For such \(\ell\), we have \(L-\ell\ge m\), and by the derivative-kernel bound \(\dot{\Sigma}^{(k)}(x,x')\le c_0\) we obtain
\[
\frac{1}{r^{L-\ell}}\prod_{k=\ell+1}^{L}\dot{\Sigma}^{(k)}(x,x')\;\le\;\left(\frac{c_0}{r}\right)^{L-\ell}.
\]
Moreover, for \(\ell\le L-m\le L-1\) the base-kernel term carries the same prefactor \(r^{-(L-\ell)}\) in \eqref{eq:ntk_explicit_form}, hence
\[
\frac{1}{r^{L-\ell}}\,\big|\Sigma^{(\ell)}(x,x')\big|\,\prod_{k=\ell+1}^{L}\dot{\Sigma}^{(k)}(x,x')
\;\le\;
C_\Sigma(x,x')\left(\frac{c_0}{r}\right)^{L-\ell}.
\]
Summing these bounds over \(\ell\le L-m\) and using a geometric-series tail gives
\[
\big|\Kop^{(L)}(x,x')-\Kop^{(L)}_{\mathrm{top},m}(x,x')\big|
\;\le\;
\bigl(1+C_\Sigma(x,x')\bigr)\sum_{j=m}^{\infty}\left(\frac{c_0}{r}\right)^j
\;=\;
\frac{1+C_\Sigma(x,x')}{1-c_0/r}\left(\frac{c_0}{r}\right)^m,
\]
as claimed. The \(\log r\) specialization follows by substituting \(m=\lceil \alpha \log r\rceil\) and observing that \((c_0/r)^{\lceil \alpha \log r\rceil}=o(r^{-p})\) for every fixed \(p>0\).
\end{proof}

\subsection{Techniques and proof sketch}\label{appendix:techniques-proof-sketch}
We highlight the main proof ingredients and how they combine.
\paragraph{Recursion and closed form.}
The RF-LR NTK recursion (Theorem~\ref{thm:ntk_recursion}) follows from a layerwise decomposition of the gradient inner product into bias, readout, and backpropagated contributions, together with conditional law of large numbers arguments under the sequential infinite-width limit~\cite{jacot2018neural}. Iterating the recursion yields the explicit \(2L\)-term expansion (Corollary~\ref{cor:general_L_layer_ntk}).

\paragraph{Microscopic decoupling and concentration.}
For three layers, the empirical kernel depends on the sample cosine and norm product of rank-\(r\) Gaussian projections. Rotational invariance yields Fisher--Kibble decoupling (Lemma~\ref{lem:fisher-kibble}), separating angular and radial fluctuations~\cite{fisher1915probable,hotelling1953new,kibble1941two}. Concentration of Gaussian norms and sample correlations then yields sub-Gaussian deviations in \(r\) for the empirical kernel (Theorem~\ref{thm:rkhs_rank_concentration} and Corollary~\ref{cor:three_layer_ntk_concentration}).

\paragraph{RKHS identification via endpoint expansions.}
The RKHS of zonal kernels on the sphere is controlled by the Puiseux behavior near \(\rho=\pm 1\)~\cite{bietti2021deepequalsshallowrelu}. We compute the mean three-layer kernel's endpoint expansion by combining Fisher's density with a hypergeometric connection formula, showing that the leading \(t^{1/2}\) exponent is preserved after taking expectations, hence the mean kernel induces the same RKHS as the shallow ReLU kernel (Corollary~\ref{thm:no_rkhs_advantage}).

\paragraph{Depth scaling and effective depth.}
Depth dependence is controlled by correlation propagation and by products of derivative kernels. In the bottleneck regime, the RF-LR recursion carries explicit \(1/r\) prefactors at each bottleneck layer, and for ReLU one has \(\sup_{\rho}\bar{\dot{\Sigma}}^{(\ell)}(\rho)\le 1/2\), which yields exponential suppression of early-layer contributions as depth increases (Section~\ref{sec:depth-scaling}). A refined mean-kernel statement is obtained by analyzing the deterministic proxy recursion (Theorem~\ref{thm:rflr-depth-scaling-mean}).

\paragraph{Future work: quadratic random-matrix scaling.}
Establishing bulk spectral laws for the \emph{empirical} RF-LR NTK in a quadratic scaling regime \(n\to\infty\) requires additional resolvent/local-law inputs. The Benigni--Paquette framework~\cite{benigni2025eigenvaluedistributionneuraltangent} establishes RMT limits for Gram spectra in the extensive-width regime (\(n\sim N^2\)) with layers taken in the RMT limit, via resolvent replacement and Gaussian-equivalence machinery for shallow two-layer NTKs. Adapting these methods to the present low-rank random-feature structure and to depth \(L\ge 2\) is left for future work.

\section{Proofs of NTK Recursions}
\label{appendix:ntk-recursions}

\subsection{Proof of Theorem~\ref{thm:ntk_recursion}: Infinite-width NTK recursion}
\label{appendix:proof-ntk-recursion}

\paragraph{Proof sketch.} Decompose the NTK at layer \(\ell\) into (i)~a bias contribution (constant \(+1\)), (ii)~Term 2: gradient inner products for the layer-\(\ell\) readouts \(A^{(\ell)}\), and (iii)~Term 1: backpropagated NTK from previous layers times the Jacobian metric. Apply the Law of Large Numbers conditionally on \(\bm{h}^{(\ell-1)}\) as \(n_\ell\to\infty\); the sequential limit ensures lower layers have already converged.
\begin{proof}
\label{proof:ntk_recursion}
We prove the recursion by decomposing the NTK at layer \( \ell \) into contributions of the trainable parameters up to layer \( \ell \), and invoking the Law of Large Numbers (LLN) under the sequential infinite-width limit. Let \( f^{(\ell)} \) denote the network output truncated at layer \( \ell \). For finite width, the (random) NTK reads
\begin{align}
\Theta^{(\ell)}_{\mathrm{rand}}(x,x') \;=\; \underbrace{\big(\nabla_{\theta^{(\ell-1)}} f^{(\ell)}(x)\big)^\top \big(\nabla_{\theta^{(\ell-1)}} f^{(\ell)}(x')\big)}_{\text{Term 1}} \;+\; \underbrace{\sum_{j=1}^{n_\ell}\frac{\partial f^{(\ell)}(x)}{\partial A^{(\ell)}_j}\frac{\partial f^{(\ell)}(x')}{\partial A^{(\ell)}_j}}_{\text{Term 2}}.
\end{align}

\paragraph{Bias parameters.}
We include a trainable layer bias \(c^{(\ell)}\) (initialized at zero). The gradient feature with respect to \(c^{(\ell)}\) is constant, so it contributes an additive constant term to the NTK; under our normalization this contribution is \(1\). This yields the additive \(+1\) term in the recursion.

\paragraph{Term 2 (layer-\(\ell\) output weights \(A^{(\ell)}\)).}
By the forward pass, \(\partial f^{(\ell)}(x)/\partial A^{(\ell)}_j = \frac{\sigma_A}{\sqrt{n_\ell}}\ReLU\!\big(\bm{w}^{(\ell)\top}_j\bm{h}^{(\ell-1)}(x)\big)\). Therefore
\begin{align}
\text{Term 2} \;=\; \frac{\sigma_A^2}{n_\ell}\sum_{j=1}^{n_\ell} \ReLU\!\big(\bm{w}^{(\ell)\top}_j\bm{h}^{(\ell-1)}(x)\big)\,\ReLU\!\big(\bm{w}^{(\ell)\top}_j\bm{h}^{(\ell-1)}(x')\big).
\end{align}
Conditionally on \(\bm{h}^{(\ell-1)}\), the summands are i.i.d. over \(\bm{w}^{(\ell)}_j\). As \(n_\ell\to\infty\) (with lower layers already taken to their limits), LLN yields
\begin{align}
\text{Term 2} \xrightarrow{p} \sigma_A^2\,\Sigma^{(\ell)}(x,x').
\end{align}

\paragraph{Term 1 (previous layers).}
By the chain rule,
\begin{align}
\nabla_{\theta^{(\ell-1)}} f^{(\ell)}(x) \;=\; \Big(\frac{\partial f^{(\ell)}(x)}{\partial \bm{h}^{(\ell-1)}(x)}\Big)\,\nabla_{\theta^{(\ell-1)}} \bm{h}^{(\ell-1)}(x).
\end{align}
Hence Term 1 factorizes as \(\Theta^{(\ell-1)}_{\mathrm{rand}}(x,x')\) multiplied by a Jacobian metric:
\begin{align}
\text{Term 1} \;=\; \Theta^{(\ell-1)}_{\mathrm{rand}}(x,x')\;\cdot\;\Big\langle \frac{\partial f^{(\ell)}}{\partial \bm{h}^{(\ell-1)}}(x), \frac{\partial f^{(\ell)}}{\partial \bm{h}^{(\ell-1)}}(x') \Big\rangle.
\end{align}
Taking the inner product and applying LLN as \(n_\ell\to\infty\) yields
\begin{align}
\Big\langle \frac{\partial f^{(\ell)}}{\partial \bm{h}^{(\ell-1)}}(x), \frac{\partial f^{(\ell)}}{\partial \bm{h}^{(\ell-1)}}(x') \Big\rangle \xrightarrow{p} \sigma_A^2\,\dot{\Sigma}^{(\ell)}(x,x').
\end{align}
Therefore,
\begin{align}
\text{Term 1} \xrightarrow{p} \Theta^{(\ell-1)}(x,x')\cdot \sigma_A^2\,\dot{\Sigma}^{(\ell)}(x,x').
\end{align}

Combining the bias contribution with the two limits completes the recursion. For bottleneck layers \( \ell \ge 2 \), the Jacobian and output contributions scale with \( 1/r \) (from the \( r \)-dimensional bottleneck); we absorb \( \sigma_A^2 \) into the normalization. Thus
\begin{align}
\Theta^{(1)}(x,x') &\;=\; 1 \;+\; \Sigma^{(1)}(x,x'),\\
\Theta^{(\ell)}(x,x') &\;=\; 1 \;+\; \frac{1}{r}\,\Theta^{(\ell-1)}(x,x')\cdot\sigma_A^2\,\dot{\Sigma}^{(\ell)}(x,x') \;+\; \frac{1}{r}\,\sigma_A^2\,\Sigma^{(\ell)}(x,x')\quad (\ell\ge2).
\end{align}
The sequential limit assumption ensures that lower-layer random objects have already converged when taking the next-layer limit, justifying the conditional LLN applications.
\end{proof}

\subsection{Proof of Theorem~\ref{thm:gp-composition}: Gaussian process composition}
\label{appendix:proof-gp-composition}

\paragraph{Proof sketch.} By induction on \(\ell\): for \(\ell=0\) the output is deterministic; for \(\ell\ge 1\), conditionally on \(\bm{h}^{(\ell-1)}\) the forward pass is a sum of i.i.d.\ terms over \(j\), so the multivariate CLT yields Gaussian convergence. The covariance matches the base kernel \(\Sigma^{(\ell)}\) by definition.
\begin{proof}
We prove by induction on \( \ell \) that \( \bm{h}^{(\ell)} \) converges to a Gaussian process with covariance kernel \( \Sigma^{(\ell)} \).

\paragraph{Base case \( \ell=0 \).}
For \( \ell=0 \), we have \( \bm{h}^{(0)}(x) = x \), which is deterministic and trivially a Gaussian process.

\paragraph{Inductive step.}
Assume that \( \bm{h}^{(\ell-1)} \) converges to a Gaussian process with covariance kernel \( \Sigma^{(\ell-1)} \). Consider the forward pass at layer \( \ell \):
\begin{align}
\bm{h}^{(\ell)}(x) = \frac{1}{\sqrt{\bm{n}_\ell}} \sum_{j=1}^{\bm{n}_\ell} A^{(\ell)}_j\,\bm{\ReLU}\!\Big(w^{(\ell)\top}_j\,\bm{h}^{(\ell-1)}(x)\Big) + c^{(\ell)}.
\end{align}

For any finite collection of inputs \( \{x_1, \dots, x_m\} \), conditionally on \( \bm{h}^{(\ell-1)} \), the terms \( A^{(\ell)}_j\,\bm{\ReLU}\!\big(w^{(\ell)\top}_j\,\bm{h}^{(\ell-1)}(x_i)\big) \) are independent across \( j \) and identically distributed. Since \( A^{(\ell)}_j \sim \mathcal{N}(0, \sigma_A^2) \) and \( w^{(\ell)}_j \) are frozen Gaussian draws, the conditional distribution of each term is centered with finite variance (and \(c^{(\ell)}\) is deterministic at initialization).

By the multivariate Central Limit Theorem, as \( \bm{n}_\ell \to \infty \), the vector \( (\bm{h}^{(\ell)}(x_1), \dots, \bm{h}^{(\ell)}(x_m)) \) converges in distribution to a multivariate Gaussian. The covariance is given by
\begin{align}
\mathbb{E}\Big[\bm{h}^{(\ell)}(x_i)\,\bm{h}^{(\ell)}(x_j)\Big] = \mathbb{E}_{w}\Big[\bm{\ReLU}\!\big(w^\top \bm{h}^{(\ell-1)}(x_i)\big)\,\bm{\ReLU}\!\big(w^\top \bm{h}^{(\ell-1)}(x_j)\big)\Big] = \Sigma^{(\ell)}(x_i, x_j),
\end{align}
where the expectation is taken over the frozen weights \( w \) and the limiting Gaussian process \( \bm{h}^{(\ell-1)} \). The last equality follows from the definition of the base kernel in Definition~\ref{def:base-derivative-kernels}.

The sequential infinite-width limit ensures that \( \bm{h}^{(\ell-1)} \) has already converged to its Gaussian process limit when taking the limit for layer \( \ell \), justifying the application of the CLT conditionally on \( \bm{h}^{(\ell-1)} \).
\end{proof}

\subsection{Proof of three-layer NTK Corollary}
\label{appendix:proof-2layer}

\paragraph{Proof sketch.} Apply the recursion (Theorem~\ref{thm:ntk_recursion}) for \(\ell=1\) and \(\ell=2\); for \(\ell=1\) use \(\Kop^{(0)}=0\); for \(\ell=2\) substitute \(\Kop^{(1)}=1+\Sigma^{(1)}\). Under homogeneity, the mean form involves expectations over Fisher (angular) and Kibble (radial); as \(r\to\infty\) these concentrate, recovering \(K_\infty\).
\begin{proof}
\label{proof:two_layer_ntk}
Apply Theorem~\ref{thm:ntk_recursion} for \( \ell=1 \) and \( \ell=2 \). For \( \ell=1 \), since \( \Kop^{(0)}=0 \), the recursion gives:
\begin{align}
\Kop^{(1)}(x,x') = 1 + \Kop^{(0)}(x,x')\cdot\dot{\Sigma}^{(1)}(x,x') + \Sigma^{(1)}(x,x') = 1 + \Sigma^{(1)}(x,x').
\end{align}
For \( \ell=2 \), substitute \( \Kop^{(1)} = 1 + \Sigma^{(1)} \) into the recursion:
\begin{align}
\Kop^{(2)}(x,x') = 1 + \frac{1}{r}\,\Kop^{(1)}(x,x')\cdot\dot{\Sigma}^{(2)}(x,x') + \frac{1}{r}\,\Sigma^{(2)}(x,x').
\end{align}
This is the stated three-layer recursion with the bias contribution.
\paragraph{Mean explicit form (low-rank).}
Under the homogeneous ReLU setting on the sphere, the mean three-layer NTK (with \( 1/r \) factors) is
\begin{align}
\tilde{\Kop}^{(2)}(\rho) \;=\; 1 \;+\; \frac{1}{r}\,\Kop^{(1)}(\rho)\,\mathbb{E}\!\left[1-\frac{\arccos(\hat{\rho}_r)}{\pi}\right] \;+\; \frac{1}{r}\,\mathbb{E}\!\left[\Sigma^{(1)}(\hat{\rho}_r)\,\|x_1\|\|y_1\|\right],
\end{align}
where the expectations are over Fisher and Kibble. As \( r\to\infty \), \( \hat{\rho}_r\to\rho \) and \( \|x_1\|\|y_1\|/r\to 1 \), so the limit \( K_\infty(\rho) = \Theta^{(1)}(\rho)(1-\arccos(\rho)/\pi) + \Sigma^{(1)}(\rho) + 1 \) is recovered (without the \( 1/r \) prefactors, since the low-rank scaling is absorbed in the limit).
\end{proof}

\subsection{Proof of General \texorpdfstring{\(L\)}{L}-layer NTK Corollary}
\label{appendix:proof-general-L}

\begin{remark}[Explicit \(2L\)-term unrolling]\label{appendix:remark-2L-expansion}
Equivalently, expanding the recursion yields an explicit \emph{linear} number of terms (namely \(2L\)). For example, for \( L=3 \):
\begin{align}
\Kop^{(3)}(x,x') &= 1 + \frac{1}{r}\dot{\Sigma}^{(2)}\cdot\dot{\Sigma}^{(3)} + \frac{1}{r}\dot{\Sigma}^{(3)} \nonumber\\
&\quad + \frac{1}{r^2}\Sigma^{(1)}\cdot\dot{\Sigma}^{(2)}\cdot\dot{\Sigma}^{(3)} + \frac{1}{r}\Sigma^{(2)}\cdot\dot{\Sigma}^{(3)} + \frac{1}{r}\Sigma^{(3)}.
\end{align}
Each bottleneck layer contributes a factor \( 1/r \); layer 1 has no such factor. In total, there are \( 2L \) terms for depth \( L \).
\end{remark}

\paragraph{Proof sketch.} By induction on \(L\): for \(L=1\) the recursion gives \(\Kop^{(1)}=1+\Sigma^{(1)}\). For \(L\ge 2\), expand \(\Kop^{(L)}\) via the recursion into a bias term and two paths (derivative-propagated and fresh-base); each path contributes \(L\) terms with the appropriate \(1/r\) factors, yielding \(2L\) terms in total.
\begin{proof}
\label{proof:general_L_ntk}
We prove the explicit form by induction on \( L \), accounting for the \( +1 \) bias terms in the recursion.

\paragraph{Base case \( L=1 \).}
From Theorem~\ref{thm:ntk_recursion}, \( \Kop^{(1)} = 1 + \Sigma^{(1)} \). The formula with \( L=1 \) gives:
\begin{align}
\Kop^{(1)}(x,x') = \sum_{\ell=1}^{1} \left(\prod_{k=\ell+1}^{1} \dot{\Sigma}^{(k)}(x,x')\right) + \sum_{\ell=1}^{1} \Sigma^{(\ell)}(x,x') \prod_{k=\ell+1}^{1} \dot{\Sigma}^{(k)}(x,x').
\end{align}
For the first sum: when \( \ell=1 \), \( \prod_{k=2}^{1} \dot{\Sigma}^{(k)} = 1 \) (empty product). For the second sum: when \( \ell=1 \), \( \Sigma^{(1)}\cdot\prod_{k=2}^{1} \dot{\Sigma}^{(k)} = \Sigma^{(1)}\cdot 1 = \Sigma^{(1)} \). Therefore, \( \Kop^{(1)} = 1 + \Sigma^{(1)} \), which matches the recursion.

\paragraph{Inductive step.}
Assume the formula holds for depth \( L-1 \):
\begin{align}
\Kop^{(L-1)}(x,x') = \sum_{\ell=1}^{L-1} \left(\prod_{k=\ell+1}^{L-1} \dot{\Sigma}^{(k)}(x,x')\right) + \sum_{\ell=1}^{L-1} \Sigma^{(\ell)}(x,x') \prod_{k=\ell+1}^{L-1} \dot{\Sigma}^{(k)}(x,x').
\end{align}
By Theorem~\ref{thm:ntk_recursion}, for \( L \ge 2 \),
\begin{align}
\Kop^{(L)}(x,x') = 1 + \frac{1}{r}\,\Kop^{(L-1)}(x,x')\cdot\dot{\Sigma}^{(L)}(x,x') + \frac{1}{r}\,\Sigma^{(L)}(x,x').
\end{align}
Substituting the inductive hypothesis and distributing the \( 1/r \) factor yields the stated closed form; the first sum gains a new term \( 1 \) (from \( \ell=L \)) and each existing term is multiplied by \( (1/r)\dot{\Sigma}^{(L)} \), while the second sum gains \( (1/r)\Sigma^{(L)} \) and each existing term is multiplied by \( (1/r)\dot{\Sigma}^{(L)} \).
The first term \( 1 \) corresponds to \( \ell=L \) in the first sum (with empty product \( \prod_{k=L+1}^{L} \dot{\Sigma}^{(k)} = 1 \)). The last term \( \Sigma^{(L)} \) corresponds to \( \ell=L \) in the second sum (with empty product \( \prod_{k=L+1}^{L} \dot{\Sigma}^{(k)} = 1 \)). Therefore:
\begin{align}
\Kop^{(L)}(x,x') = \sum_{\ell=1}^{L} \left(\prod_{k=\ell+1}^{L} \dot{\Sigma}^{(k)}(x,x')\right) + \sum_{\ell=1}^{L} \Sigma^{(\ell)}(x,x') \prod_{k=\ell+1}^{L} \dot{\Sigma}^{(k)}(x,x'),
\end{align}
completing the induction. The total number of terms is \( L + L = 2L \), which grows linearly with depth.
\end{proof}

\subsection{Reference: full-width MLP limiting NTK at the edge of chaos}
\label{appendix:mlp-eoc-ntk}
\begin{proposition}[Closed form for full-width MLP EOC NTK]\label{prop:mlp-eoc-ntk-formula}
Let \(l\ge 1\) and consider an infinitely wide \(l\)-layer MLP with \((a,b)\)-ReLU activation initialized at the edge of chaos as in~\cite{terjek2025mlpsateoc}. Denote by \(\varrho\) the corresponding cosine map and by \(\rho_1(x,x')\) the input cosine similarity. Then the limiting NTK satisfies

\]
with the convention that the empty product (for \(k=l\)) equals \(1\). See~\cite{terjek2025mlpsateoc} for the derivation.
\end{proposition}



\section{Proofs of low-rank NTK RKHS}

\subsection{Background and statements for Section~\ref{sec:no-rkhs-advantage}}\label{appendix:rkhs-statements}
\paragraph{RKHS viewpoint in the NTK regime.}
In the NTK regime, training dynamics linearize:
\[
f_t(x) \approx f_0(x) + \langle \nabla_\theta f_0(x),\, \theta_t-\theta_0 \rangle.
\]
The learned function stays in the RKHS induced by the (limiting) kernel \(\Theta\), with norm
\[
\|f\|_{\mathcal{H}_\Theta}^2 = \langle f, \Theta^{-1}f\rangle_{L^2}.
\]
For RF-LR with frozen random features, universal approximation holds due to full support of the frozen Gaussian draws \(w^{(\ell)}\), so any function in the Barron space~\cite{e2021barronspaceflowinducedfunction} can be approximated with error \(O(1/\sqrt{N})\).
Classical NTK analysis suggests that feature weights move little during training; RF-LR enforces this by design by freezing the feature weights, which isolates kernel-regime behavior without an additional weight-deviation analysis.

\paragraph{Mean RKHS and random features.}
Each realization \(\omega\) of the random features yields a kernel \(K_\omega(x,x')=\Theta^{(2)}_\omega(x,x')\) and an induced RKHS \(\mathcal{H}_\omega\). The mean kernel \(\tilde{K}(x,x')=\mathbb{E}_\omega[K_\omega(x,x')]\) is deterministic; its RKHS \(\mathcal{H}_{\tilde{K}}\) is the RKHS of \(\tilde{K}\). It is not \(\mathbb{E}[\mathcal{H}_\omega]\) (expectation of Hilbert spaces is undefined). For each \(\omega\), the feature map \(\varphi_\omega(x)=K_\omega(\cdot,x)\in\mathcal{H}_\omega\) is random, whereas the canonical feature map of the mean kernel \(\varphi_{\tilde{K}}(x)=\tilde{K}(\cdot,x)\in\mathcal{H}_{\tilde{K}}\) is deterministic. The mean RKHS describes the typical function space obtained after averaging over random feature realizations.

\paragraph{Remark (application of Bietti--Bach).}\label{appendix:rmk-bietti-bach-application}
Once the mean kernel's Puiseux endpoint behavior is established (Appendix~\ref{appendix:proof-conditional-rkhs-homogeneous}), the RKHS equivalence in Corollary~\ref{thm:no_rkhs_advantage} follows directly from the characterization theorem of~\cite{bietti2021deepequalsshallowrelu}.

\begin{corollary}[Mean NTK over Fisher--Kibble has same RKHS: three-layer case]\label{cor:conditional_rkhs_homogeneous}
Under Assumption~\ref{assump:homogeneous_activation} and the RF-LR setting with ReLU nonlinearity and isotropic random features on \(\mathbb{S}^{d-1}\), the mean three-layer NTK \(\tilde{\Kop}^{(2)}\) (obtained by taking expectations over Fisher and Kibble distributions) is a zonal kernel that induces the same RKHS as the shallow ReLU kernel. In particular, the RKHSs coincide as sets with equivalent norms.
\end{corollary}
Proof: See Appendix~\ref{appendix:proof-conditional-rkhs-homogeneous}.

\paragraph{Main RKHS result (stated in the main text).}
Corollary~\ref{thm:no_rkhs_advantage} is stated in Section~\ref{sec:no-rkhs-advantage}. It follows by combining the Puiseux endpoint analysis in Appendix~\ref{appendix:proof-conditional-rkhs-homogeneous} with the RKHS characterization of~\cite{bietti2021deepequalsshallowrelu}.




\subsection{Discussion on Fisher and Kibble Distributions}
\label{appendix:fisher-kibble}

\begin{lemma}[Fisher–Kibble decoupling]
\label{lem:fisher-kibble-appendix}
Let \(x,y\) be input vectors and let \(x_1,y_1\) denote their rank-\(r\) random projections. Define the empirical correlation \( \rho_1 = \cos\angle(x_1,y_1) \) and squared norms \( u=\|x_1\|^2, v=\|y_1\|^2 \). Then:
\begin{itemize}
    \item \( \rho_1 \) follows Fisher's correlation distribution~\cite{fisher1915probable} \cite{hotelling1953new}, centered at the true correlation \( \rho \).
    \item \( (u,v) \) follow Kibble's~\cite{kibble1941two} bivariate Gamma (chi-square) law.
    \item Angular and radial parts are independent: \( p(\rho_1,u,v) = p_{\text{Fisher}}(\rho_1)\,p_{\text{Kibble}}(u,v) \).
\end{itemize}
\end{lemma}

\begin{remark}[Fisher--Kibble decoupling and independence]\label{rmk:fisher-kibble-law}
By rotational invariance, the empirical correlation and squared norms satisfy
\[
\rho_1 \sim \mathrm{Fisher}(\rho,r),\qquad (u,v) \sim \mathrm{Kibble}(r,\rho),
\qquad\text{and}\qquad p(\rho_1,u,v)=p_{\mathrm{Fisher}}(\rho_1)\,p_{\mathrm{Kibble}}(u,v).
\]
\paragraph{Homogeneity and factorization (why angular and radial parts decouple).}\label{appendix:homogeneity-decoupling}
For positively 1-homogeneous activations (Assumption~\ref{assump:homogeneous_activation}), the first-layer base kernel separates radial and angular dependence: for all \(x,x'\neq 0\),
\[
\Sigma^{(1)}(x,x')=\|x\|\,\|x'\|\,\tilde{\Sigma}^{(1)}(\rho),\qquad \rho=\frac{\langle x,x'\rangle}{\|x\|\,\|x'\|},
\]
where \(\tilde{\Sigma}^{(1)}\) is a scalar function on \([-1,1]\) (for ReLU it is the standard arc-cosine kernel). Under isotropic Gaussian projections, rotational invariance implies that the empirical correlation \(\rho_1=\cos\angle(x_1,y_1)\) depends only on the angular part of the projected pair, while the norms \((\|x_1\|,\|y_1\|)\) depend only on the radial part; consequently \(\rho_1\) is independent of \((u,v)=(\|x_1\|^2,\|y_1\|^2)\) and the joint law factorizes as stated above.
The Fisher distribution density~\cite{fisher1915probable} is
\[
p(\rho_1 \mid \rho, r)
=
\frac{(r-2)\,\Gamma(r-1)\,(1-\rho^2)^{\frac{r-1}{2}}\,(1-\rho_1^2)^{\frac{r-4}{2}}}{\sqrt{2\pi}\,\Gamma(r-\tfrac{1}{2})\,(1-\rho\rho_1)^{r-\frac{3}{2}}}\,
{}_2F_1\!\left(\tfrac{1}{2},\tfrac{1}{2};\, r-\tfrac{1}{2};\, \tfrac{1+\rho\rho_1}{2}\right),
\qquad r>2.
\]
\medskip
\noindent
For large \(r\), the inverse Gudermann function applied to the complementary angle is approximately normal:
\[
\mathrm{gd}^{-1}\!\left(\frac{\pi}{2}-\arccos(\rho_1)\right)
\approx
\mathcal{N}\!\left(\mathrm{arctanh}(\rho),\,\frac{1}{r-3}\right),
\qquad\text{hence}\qquad \mathrm{Var}(\rho_1)=O(1/r).
\]
\medskip
\noindent
Kibble's~\cite{kibble1941two} bivariate chi-square law for \((u,v)=(\|x_1\|^2,\|y_1\|^2)\) has density
\[
f(u,v)
=
\frac{(uv)^{\frac{r/2-1}{2}}\,\exp\!\left(-\tfrac{u+v}{2(1-\rho^2)}\right)}{\Gamma(r/2)\,\big(2(1-\rho^2)\big)^{\frac{r}{2}+1}\,\rho^{\frac{r/2-1}{2}}}\,
I_{\frac{r}{2}-1}\!\left(\frac{\rho\sqrt{uv}}{1-\rho^2}\right),
\qquad u,v\ge 0,
\]
where \(I_\nu\) is the modified Bessel function of the first kind. The moments are
\[
\mathbb{E}[u]=\mathbb{E}[v]=r,\qquad \mathrm{Var}(u)=\mathrm{Var}(v)=2r,\qquad \mathrm{Cov}(u,v)=2r\,\rho^2.
\]
Defining \(w_r=\sqrt{uv}/r\), concentration results imply \(\mathbb{E}[w_r]\to 1\) and \(\mathrm{Var}(w_r)=O(1/r)\).

\paragraph{Compact three-layer empirical NTK representation.}
In the three-layer low-rank setting, denote
\[
\rho_1=\cos\angle(\bm{h}^{(1)}(x),\bm{h}^{(1)}(x')),\qquad
w_r=\frac{\|\bm{h}^{(1)}(x)\|\,\|\bm{h}^{(1)}(x')\|}{r}.
\]
Then the empirical three-layer NTK admits the compact form
\[
\Kop^{(2)}(x,x')
=
1
+\frac{1}{r}\,\Kop^{(1)}(\rho_1)\!\left(1-\frac{\arccos(\rho_1)}{\pi}\right)
+\frac{1}{r}\,w_r\,\Sigma^{(1)}(\rho_1),
\]
with \( \Sigma^{(1)}(u)=\frac{1}{\pi}\big(\sqrt{1-u^2}+u(1-\arccos u)\big) \).
\end{remark}

\paragraph{Proof sketch.} The rank-\(r\) projections \(x_1=Px\), \(y_1=Py\) form a \(\rho\)-correlated Gaussian pair. By rotational invariance, the empirical correlation \(\rho_1=\cos\angle(x_1,y_1)\) (angular part) and the squared norms \((u,v)=(\|x_1\|^2,\|y_1\|^2)\) (radial part) are independent. The angular part follows Fisher's distribution; the radial part follows Kibble's bivariate chi-square law.
\begin{proof}
\label{proof:fisher_kibble}
Let \(x, y \in \mathbb{R}^d\) be fixed input vectors with correlation \(\rho = \langle x, y\rangle/(\|x\|\|y\|)\). Consider a rank-\(r\) random projection matrix \(P \in \mathbb{R}^{r \times d}\) with entries \(P_{ij} \sim \mathcal{N}(0, 1/d)\) i.i.d., so that \(x_1 = Px\) and \(y_1 = Py\) are the projected vectors.

Since \(P\) has i.i.d. Gaussian entries, the projected vectors \(x_1, y_1 \in \mathbb{R}^r\) form a \(\rho\)-correlated bivariate Gaussian pair. Specifically, for each component \(i = 1, \dots, r\), the pairs \((x_{1i}, y_{1i})\) are i.i.d. with joint distribution
\begin{align}
\begin{pmatrix} x_{1i} \\ y_{1i} \end{pmatrix} \sim \mathcal{N}\left(\mathbf{0}, \begin{pmatrix} \|x\|^2/d & \rho \|x\|\|y\|/d \\ \rho \|x\|\|y\|/d & \|y\|^2/d \end{pmatrix}\right).
\end{align}
By rotational invariance of the Gaussian projection and the fact that the correlation \(\rho\) is preserved under orthogonal transformations, we can assume without loss of generality that the covariance structure factors into angular and radial components.

\paragraph{Angular part (Fisher distribution).}
The empirical correlation is defined as
\begin{align}
\rho_1 = \frac{\langle x_1, y_1 \rangle}{\|x_1\|\|y_1\|} = \frac{\sum_{i=1}^r x_{1i} y_{1i}}{\sqrt{\sum_{i=1}^r x_{1i}^2}\sqrt{\sum_{i=1}^r y_{1i}^2}}.
\end{align}
This is the sample correlation coefficient of \(r\) pairs of correlated Gaussian random variables. Fisher~\cite{fisher1915probable,hotelling1953new} showed that the distribution of \(\rho_1\) depends only on the true correlation \(\rho\) and the sample size \(r\), with density given in Remark~\ref{rmk:fisher-kibble-law}. The key observation is that \(\rho_1\) is a function purely of the angles between the projected vectors, independent of their magnitudes.

\paragraph{Radial part (Kibble distribution).}
The squared norms \(u = \|x_1\|^2 = \sum_{i=1}^r x_{1i}^2\) and \(v = \|y_1\|^2 = \sum_{i=1}^r y_{1i}^2\) are sums of correlated chi-square variables. Since \((x_{1i}, y_{1i})\) are \(\rho\)-correlated Gaussians, the bivariate distribution of \((u, v)\) follows Kibble's~\cite{kibble1941two} bivariate chi-square law. The joint density involves the modified Bessel function \(I_{\nu}(z)\), which appears because the dot product \(\langle x_1, y_1 \rangle = \sum_{i=1}^r x_{1i} y_{1i}\) can be expressed as a weighted sum of products of correlated normals, whose distribution relates to Bessel functions through the generating function of the bivariate chi-square distribution.

Specifically, the joint characteristic function of \((u, v)\) is
\begin{align}
\phi(s, t) = \mathbb{E}[e^{i(su + tv)}] = \left(1 - 2i(1-\rho^2)(s+t) - 4\rho^2 st\right)^{-r/2},
\end{align}
and the inverse Fourier transform yields Kibble's density, which contains the modified Bessel function \(I_{\frac{r}{2}-1}\) as stated in Remark~\ref{rmk:fisher-kibble-law}. The Bessel function arises from the integral representation of the bivariate chi-square density when the correlation \(\rho \neq 0\).

\paragraph{Independence of angular and radial parts.}
Rotational invariance of isotropic Gaussian projections implies that \(\rho_1 = \cos\angle(x_1, y_1)\) and the norms \((\|x_1\|, \|y_1\|)\) are independent, yielding the factorization \(p(\rho_1, u, v) = p_{\text{Fisher}}(\rho_1) \cdot p_{\text{Kibble}}(u, v)\). This is the classical independence of the sample correlation from sample variances in bivariate normal data.
\end{proof}

\subsection{Open directions: deep mean NTK recursion and effective depth}\label{appendix:open-directions}
The deterministic proxy (Definition~\ref{def:deterministic-proxy}) analyzes the scalar recursion along the mean path \( \rho_k = \varrho^{\circ(k-1)}(\rho_1) \). The actual network uses the \emph{random} correlation chain \( \rho_\ell \) with \( O(1/\sqrt{r}) \) per-layer fluctuations (Appendix~\ref{appendix:random-correlation-chain}). A rigorous link requires: (i)~propagating the Fisher-type concentration \( \mathbb{E}[(\rho_\ell - \varrho^{\circ(\ell-1)}(\rho_1))^2] \le C/r \) through depth to show \( \Theta^{(L)}_{\mathrm{random}} - \Theta^{(L)}_{\mathrm{proxy}} = O_P(\sqrt{L/r}) \) or similar; (ii)~a rigorous recursion for \( \tilde{\Kop}^{(\ell)} \) with explicit covariance bounds; (iii)~extension of Fisher--Kibble decoupling to nested layers to control mixed terms such as \( \mathbb{E}[\Kop^{(\ell-1)}\dot{\Sigma}^{(\ell)}] \) (where dependence through the deep GP chain prevents naive factorization). The exponential suppression of early-layer contributions suggests that very deep MMNNs may exhibit an ``effective depth'' phenomenon similar to ResNets, where only the top layers meaningfully contribute to the NTK.

\begin{lemma}[Nonasymptotic concentration of Gaussian sample cosine]\label{lem:gaussian-sample-cosine}
Let \(r\ge 1\) and let \((X,Y)\in\mathbb{R}^r\times\mathbb{R}^r\) have i.i.d.\ coordinates \(\{(X_i,Y_i)\}_{i=1}^r\), each distributed as a centered bivariate normal with \(\mathbb{E}[X_i^2]=\mathbb{E}[Y_i^2]=1\) and \(\mathbb{E}[X_iY_i]=\rho\in(-1,1)\).
Define the empirical cosine (sample correlation without centering)
\[
\widehat{\rho}_r \;=\; \frac{\langle X,Y\rangle}{\|X\|\,\|Y\|}.
\]
Then there exist absolute constants \(c,C>0\) such that for all \(t\in(0,1)\),
\[
\mathbb{P}\big(|\widehat{\rho}_r-\rho|\ge t\big)\;\le\;6\exp(-c\,r\,t^2),
\qquad
\mathbb{E}\big[(\widehat{\rho}_r-\rho)^2\big]\;\le\;\frac{C}{r}.
\]
\end{lemma}
\paragraph{Proof sketch.} Write \(\widehat{\rho}_r=S_{xy}/\sqrt{S_{xx}S_{yy}}\) where \(S_{xx},S_{yy},S_{xy}\) are sample averages. Each of \(S_{xx}-1\), \(S_{yy}-1\), \(S_{xy}-\rho\) is sub-exponential; Bernstein's inequality yields tail bounds. On the event that \(S_{xx},S_{yy}\) are close to 1, bound \(|\widehat{\rho}_r-\rho|\) via a ratio expansion; union bound over the three tails gives the stated rate.
\begin{proof}
Write
\[
S_{xx}=\frac{1}{r}\|X\|^2,\qquad S_{yy}=\frac{1}{r}\|Y\|^2,\qquad S_{xy}=\frac{1}{r}\langle X,Y\rangle,
\qquad\text{so that}\qquad
\widehat{\rho}_r=\frac{S_{xy}}{\sqrt{S_{xx}S_{yy}}}.
\]
Each of \(S_{xx}-1\), \(S_{yy}-1\), and \(S_{xy}-\rho\) is an average of i.i.d.\ centered sub-exponential random variables (since \(X_i^2-1\), \(Y_i^2-1\), and \(X_iY_i-\rho\) have finite \(\psi_1\) norms for Gaussian data). Therefore, by Bernstein's inequality, there exist absolute constants \(c_0>0\) and \(C_0<\infty\) such that for all \(t\in(0,1)\),
\[
\mathbb{P}\big(|S_{xx}-1|\ge t\big)\le 2e^{-c_0 r t^2},\quad
\mathbb{P}\big(|S_{yy}-1|\ge t\big)\le 2e^{-c_0 r t^2},\quad
\mathbb{P}\big(|S_{xy}-\rho|\ge t\big)\le 2e^{-c_0 r t^2}.
\]
On the event \(E_t=\{|S_{xx}-1|\le t,\ |S_{yy}-1|\le t\}\) with \(t\le 1/2\), we have \(S_{xx},S_{yy}\in[1/2,3/2]\), hence
\[
\Bigl|\frac{1}{\sqrt{S_{xx}S_{yy}}}-1\Bigr|
\le C_0\big(|S_{xx}-1|+|S_{yy}-1|\big)
\le 2C_0 t
\]
for an absolute constant \(C_0\). Using
\[
\widehat{\rho}_r-\rho
=\frac{S_{xy}-\rho}{\sqrt{S_{xx}S_{yy}}}
\;+\;\rho\Bigl(\frac{1}{\sqrt{S_{xx}S_{yy}}}-1\Bigr),
\]
we obtain on \(E_t\) the bound \(|\widehat{\rho}_r-\rho|\le C_1(|S_{xy}-\rho|+t)\) for an absolute constant \(C_1\). Taking \(t\asymp t'\) and applying a union bound over the three Bernstein tails yields \(\mathbb{P}(|\widehat{\rho}_r-\rho|\ge t')\le 6e^{-c r t'^2}\) for \(t'\in(0,1)\) and absolute \(c>0\).
The second-moment bound follows by integrating the tail bound: \(\mathbb{E}[(\widehat{\rho}_r-\rho)^2]=\int_0^\infty 2u\,\mathbb{P}(|\widehat{\rho}_r-\rho|\ge u)\,du\le C/r\).
\end{proof}

\subsection{Concentration of the random Gram matrix around the proxy}\label{appendix:proxy-empirical-concentration}
Let \(\hat{K}\) be the \(n\times n\) random Gram matrix with entries \(\hat{K}_{ij}=\hat{\Theta}^{(L)}_r(x_i,x_j)\) (the empirical RF-LR NTK over the random correlation chain), and let \(K_{\mathrm{proxy}}\) be the deterministic proxy Gram matrix with entries \((K_{\mathrm{proxy}})_{ij}=\Theta^{(L)}(\rho_{1,ij})\) (Definition~\ref{def:deterministic-proxy}). Under the same per-layer sub-Gaussian concentration as in Appendix~\ref{appendix:random-correlation-chain},
\[
\mathbb{E}\bigl[(\rho_\ell-\varrho^{\circ(\ell-1)}(\rho_1))^2\bigr]\le C/r,\qquad
\mathbb{P}\bigl(|\rho_\ell-\varrho^{\circ(\ell-1)}(\rho_1)|\ge t\bigr)\le 6\exp(-c r t^2),
\]
the kernel recursion is Lipschitz in the correlation path. Propagating the per-step bounds through \(L\) layers (e.g.\ by a union bound over the \(L\) steps and the fact that \(\Theta^{(k)}\) is Lipschitz in \(\rho\) with constant \(O(1)\)), one obtains that for each pair \((i,j)\),
\[
\bigl|\hat{K}_{ij}-(K_{\mathrm{proxy}})_{ij}\bigr|=O_P(\sqrt{L/r}).
\]
A union over the \(n^2\) pairs and the inequality \(\|A\|_{\mathrm{op}}\le \|A\|_{\mathrm{F}}\le n\max_{i,j}|A_{ij}|\) yield the following. The bound in Proposition~\ref{prop:proxy-empirical-concentration} is \emph{sketch-based}; a full proof with explicit constants and rate is given only for the equicorrelated case (Theorem~\ref{thm:equicorrelated-op-bound}).
\begin{proposition}[Proxy--empirical Gram matrix concentration]\label{prop:proxy-empirical-concentration}
Fix \(n,L\ge 1\), \(r>1\), and inputs \(x_1,\ldots,x_n\) with pairwise cosine similarities \(\rho_{1,ij}\in(-1,1)\). Let \(\hat{K}\) and \(K_{\mathrm{proxy}}\) be as above. There exist constants \(C,c>0\) (depending on \(L,n\) and the \(\rho_{1,ij}\)) such that for any \(\epsilon\in(0,1)\),
\[
\mathbb{P}\!\left(\|\hat{K}-K_{\mathrm{proxy}}\|_{\mathrm{op}}\;\ge\; \epsilon\right)\;\le\; C\,\exp\!\left(-c\,\frac{r\,\epsilon^2}{L\,n^2}\right).
\]
In particular, for \(r\to\infty\) with \(n,L\) fixed, \(\|\hat{K}-K_{\mathrm{proxy}}\|_{\mathrm{op}}=o_P(1)\). If in addition the proxy Gram matrix has minimum eigenvalue \(\lambda_{\min}(K_{\mathrm{proxy}}|_{\mathbf{1}^\perp})\ge \gamma>0\), then for \(r\) large enough so that \(\|\hat{K}-K_{\mathrm{proxy}}\|_{\mathrm{op}}<\gamma/2\) with high probability, Weyl's inequality gives
\[
\bigl|\lambda_{\min}(\hat{K}|_{\mathbf{1}^\perp})-\lambda_{\min}(K_{\mathrm{proxy}}|_{\mathbf{1}^\perp})\bigr|\le \gamma/2,
\]
so the condition number of \(\hat{K}\) restricted to \(\mathbf{1}^\perp\) is within a constant factor of that of \(K_{\mathrm{proxy}}\).
\end{proposition}
\paragraph{Proof sketch.} Per-layer concentration (Lemma~\ref{lem:gaussian-sample-cosine}) and Lipschitz dependence of the scalar recursion on \(\rho\) imply that each entry \(\hat{K}_{ij}\) differs from \((K_{\mathrm{proxy}})_{ij}\) by \(O_P(\sqrt{L/r})\). Union bound over \(n^2\) pairs with sub-Gaussian tails gives
\[
\max_{i,j}\bigl|\hat{K}_{ij}-(K_{\mathrm{proxy}})_{ij}\bigr|=O_P\bigl(\sqrt{L\log(n)/r}\bigr).
\]
Hence \(\|\hat{K}-K_{\mathrm{proxy}}\|_{\mathrm{op}}\le n\|\hat{K}-K_{\mathrm{proxy}}\|_{\max}=O_P(n\sqrt{L\log(n)/r})\), and the stated exponential tail follows. The condition-number comparison is by Weyl's inequality for the eigenvalues of symmetric matrices. For general (non-equicorrelated) datasets this remains a proof sketch; a rigorous operator-norm bound with explicit rate is given only for the equicorrelated case in Theorem~\ref{thm:equicorrelated-op-bound}.

\paragraph{Equicorrelated case: reduction to two scalars and rigorous bound on \(\mathbf{1}^\perp\).}
In the equicorrelated case (\(\rho_{1,ij}=\rho_0\) for all \(i\neq j\), \(\rho_{1,ii}=1\)), the proxy Gram matrix has the form \(K_{\mathrm{proxy}}=\Theta^{(L)}(1)I_n+\Theta^{(L)}(\rho_0)(\mathbf{1}\mathbf{1}^\top-I_n)\). By symmetry of the data and the network, the random \(\hat{K}\) has the same structure: \(\hat{K}_{ii}=\hat{K}_{11}\) and \(\hat{K}_{ij}=\hat{K}_{12}\) for \(i\neq j\). Hence the deviation matrix \(E\mathrel{:=}\hat{K}-K_{\mathrm{proxy}}\) is
\[
E \;=\; \bigl(\hat{K}_{11}-\Theta^{(L)}(1)\bigr)\,I_n \;+\; \bigl(\hat{K}_{12}-\Theta^{(L)}(\rho_0)\bigr)\,\bigl(\mathbf{1}\mathbf{1}^\top-I_n\bigr),
\]
with eigenvalues \(\lambda_{\mathbf{1}}=\bigl(\hat{K}_{11}-\Theta^{(L)}(1)\bigr)+(n-1)\bigl(\hat{K}_{12}-\Theta^{(L)}(\rho_0)\bigr)\) on \(\mathbf{1}\) and \(\lambda_\perp=\bigl(\hat{K}_{11}-\Theta^{(L)}(1)\bigr)-\bigl(\hat{K}_{12}-\Theta^{(L)}(\rho_0)\bigr)\) on \(\mathbf{1}^\perp\) (multiplicity \(n-1\)), so \(\|\hat{K}-K_{\mathrm{proxy}}\|_{\mathrm{op}}=\max(|\lambda_{\mathbf{1}}|,|\lambda_\perp|)\). The following theorem gives a rigorous \(O_P(L/r)\) bound for the restriction to \(\mathbf{1}^\perp\); the full operator norm without \(n\)-dependence remains open (see Remark~\ref{rmk:equicorrelated-full-op}).

\begin{lemma}[Path-wise kernel and sensitivity]\label{lem:path-kernel-sensitivity}
Let \(s(\rho)\) and \(\dot{s}(\rho)\) be the scalar ReLU base and derivative kernels as in \eqref{eq:rflr-scalar-mean-ntk-rec}. For \(k\ge 1\) and \(\rho_1,\ldots,\rho_k\in(-1,1)\), define \(\Phi^{(1)}(\rho_1)=1+s(\rho_1)\) and for \(k\ge 2\),
\[
\Phi^{(k)}(\rho_1,\ldots,\rho_k) \;=\; 1 \;+\; \frac{1}{r}\,\Phi^{(k-1)}(\rho_1,\ldots,\rho_{k-1})\,\dot{s}(\rho_k) \;+\; \frac{1}{r}\,s(\rho_k).
\]
Then \(\Phi^{(k)}\) is the same function as \(\Theta^{(k)}\) when evaluated along the deterministic path \(\rho_j=\varrho^{\circ(j-1)}(\rho_1)\). There exist constants \(C_s,C_\Phi<\infty\) (depending only on the ReLU kernels) such that for all \(2\le \ell\le k\le L\) and all \(\rho_1,\ldots,\rho_k\in[-1+\delta,1-\delta]\) with \(\delta\in(0,1)\),
\[
\left|\frac{\partial\Phi^{(k)}}{\partial\rho_\ell}\right|
\;\le\; C_\Phi\,\left(\frac{1}{r}\right)^{k-\ell}.
\]
In particular, for \(r\ge 2\) and some \(\overline{C}_L,\overline{C}_L''=O(1)\),
\begin{align*}
\sum_{\ell=2}^L \left(\frac{\partial\Phi^{(L)}}{\partial\rho_\ell}\right)^{\!2} &\le C_\Phi^2\,\frac{1-(1/r)^{2(L-1)}}{1-(1/r)^2} \le \overline{C}_L, \\
\sum_{\ell=2}^L \left|\frac{\partial\Phi^{(L)}}{\partial\rho_\ell}\right| &\le C_\Phi\sum_{t=0}^{L-2}(1/r)^t \le \overline{C}_L''.
\end{align*}
\end{lemma}
\begin{proof}
The recursion for \(\Phi^{(k)}\) coincides with \eqref{eq:rflr-scalar-mean-ntk-rec}, so
\[
\Theta^{(k)}(\rho_1) \;=\; \Phi^{(k)}\bigl(\rho_1,\varrho(\rho_1),\ldots,\varrho^{\circ(k-1)}(\rho_1)\bigr).
\]
Differentiating in \(\rho_k\),
\[
\frac{\partial\Phi^{(k)}}{\partial\rho_k} \;=\; \frac{1}{r}\,\Phi^{(k-1)}\,\dot{s}'(\rho_k) \;+\; \frac{1}{r}\,s'(\rho_k).
\]
On \([-1+\delta,1-\delta]\), \(|\dot{s}'|\) and \(|s'|\) are bounded by some \(C_s\). Since \(\Phi^{(k-1)}\) is bounded (the recursion yields \(\Phi^{(k)}\in[1,1+O(1)]\) on compact \(\rho\)), we have \(\bigl|\frac{\partial\Phi^{(k)}}{\partial\rho_k}\bigr|\le C_s'(1/r)\) for a constant \(C_s'\). For \(\ell<k\),
\[
\frac{\partial\Phi^{(k)}}{\partial\rho_\ell} \;=\; \frac{1}{r}\,\frac{\partial\Phi^{(k-1)}}{\partial\rho_\ell}\,\dot{s}(\rho_k).
\]
By induction,
\[
\left|\frac{\partial\Phi^{(k)}}{\partial\rho_\ell}\right| \;\le\; C_s'\,\prod_{j=\ell+1}^k \frac{1}{r}|\dot{s}(\rho_j)| \;\le\; C_\Phi\,(1/r)^{k-\ell},
\]
with \(C_\Phi=C_s'\,(\sup|\dot{s}|)^{L-1}\le C_s'/2^{L-1}\). The sum bound follows from \(\sum_{\ell=2}^L (1/r)^{2(L-\ell)}=\sum_{t=0}^{L-2}(1/r)^{2t}\le \frac{1}{1-(1/r)^2}\le 2\) for \(r\ge 2\), so \(\sum_{\ell=2}^L \bigl(\frac{\partial\Phi^{(L)}}{\partial\rho_\ell}\bigr)^2\le C_\Phi^2\cdot 2=\overline{C}_L\).
\end{proof}

\begin{lemma}[Second-order remainder]\label{lem:second-order-remainder}
Under the same setting as Lemma~\ref{lem:path-kernel-sensitivity}, let \(\rho_1=\rho_0\) be fixed and \((\rho_2,\ldots,\rho_L)\) be the random correlation chain with \(\mathbb{E}[(\rho_\ell-\varrho^{\circ(\ell-1)}(\rho_0))^2]\le C_0/r\) and \(\mathbb{P}(|\rho_\ell-\varrho^{\circ(\ell-1)}(\rho_0)|\ge t)\le 6\exp(-c_0 r t^2)\) for all \(\ell\) (Lemma~\ref{lem:gaussian-sample-cosine}). Let \(\bar{\rho}_\ell=\varrho^{\circ(\ell-1)}(\rho_0)\). Then the second-order Taylor remainder
\[
R_2 \;=\; \hat{\Theta}^{(L)}_r - \Theta^{(L)}(\rho_0) \;-\; \sum_{\ell=2}^L \frac{\partial\Phi^{(L)}}{\partial\rho_\ell}(\bar{\rho}_2,\ldots,\bar{\rho}_L)\,\bigl(\rho_\ell-\bar{\rho}_\ell\bigr)
\]
satisfies \(\mathbb{E}[|R_2|]\le B_L\,L/r\) for a constant \(B_L\) depending on \(L\) and the Hessian of \(\Phi^{(L)}\) on \([-1+\delta,1-\delta]^L\), and for any \(u>0\),
\[
\mathbb{P}(|R_2|\ge u) \;\le\; 2\exp\!\left(-\min\!\left(\frac{c_0 r u}{2B_L L},\,\frac{c_0 r u^2}{2(B_L L)^2}\right)\right).
\]
\end{lemma}
\begin{proof}
By Taylor's theorem, for some \(\xi\) on the segment between \((\rho_2,\ldots,\rho_L)\) and \((\bar{\rho}_2,\ldots,\bar{\rho}_L)\),
\[
R_2 \;=\; \frac{1}{2}\sum_{\ell,m} \frac{\partial^2\Phi^{(L)}}{\partial\rho_\ell\partial\rho_m}(\xi)\,(\rho_\ell-\bar{\rho}_\ell)(\rho_m-\bar{\rho}_m).
\]
The second derivatives of \(\Phi^{(L)}\) are bounded on compact sets (the recursion is polynomial in \(s,\dot{s}\) and those are \(C^\infty\) on \((-1,1)\)). Hence
\[
|R_2| \;\le\; \frac{1}{2}H_L \sum_{\ell=2}^L (\rho_\ell-\bar{\rho}_\ell)^2
\]
for a constant \(H_L\), and
\[
\mathbb{E}[|R_2|] \;\le\; \frac{1}{2}H_L\,\sum_{\ell=2}^L \mathbb{E}[(\rho_\ell-\bar{\rho}_\ell)^2] \;\le\; \frac{1}{2}H_L\,(L-1)\,\frac{C_0}{r} \;\le\; B_L\,\frac{L}{r}.
\]
Each \((\rho_\ell-\bar{\rho}_\ell)^2\) is sub-exponential (from the sub-Gaussian tail of \(\rho_\ell-\bar{\rho}_\ell\)). By Bernstein's inequality for sums of sub-exponential variables, \(\sum_{\ell=2}^L (\rho_\ell-\bar{\rho}_\ell)^2\) has an exponential tail with scale \(O(L/r)\); hence \(|R_2|\) has the stated tail bound.
\end{proof}

\paragraph{Proof of Theorem~\ref{thm:equicorrelated-op-bound}.}
\begin{proof}
Let \(\hat{K}\) and \(K_{\mathrm{proxy}}\) be as in Proposition~\ref{prop:proxy-empirical-concentration}. Let \(E=\hat{K}-K_{\mathrm{proxy}}\). We have
\[
\|E|_{\mathbf{1}^\perp}\|_{\mathrm{op}} \;=\; |\lambda_\perp| \;=\; \bigl|\hat{K}_{11}-\Theta^{(L)}(1)-\bigl(\hat{K}_{12}-\Theta^{(L)}(\rho_0)\bigr)\bigr|.
\]
For the diagonal, \(\rho_1=1\) is fixed and the random path \((1,\rho_2^{\mathrm{diag}},\ldots,\rho_L^{\mathrm{diag}})\) has the same per-layer concentration; so \(\hat{K}_{11}-\Theta^{(L)}(1)\) admits the same Taylor expansion and bounds as below with \(\rho_0\) replaced by \(1\). Thus it suffices to prove that both \(|\hat{K}_{11}-\Theta^{(L)}(1)|\) and \(|\hat{K}_{12}-\Theta^{(L)}(\rho_0)|\) are \(O_P(L/r)\); then
\[
|\lambda_\perp| \;\le\; \bigl|\hat{K}_{11}-\Theta^{(L)}(1)\bigr| + \bigl|\hat{K}_{12}-\Theta^{(L)}(\rho_0)\bigr| \;=\; O_P(L/r).
\]

\medskip
\noindent\textbf{Off-diagonal deviation.} Write \(\hat{K}_{12}=\Phi^{(L)}(\rho_0,\rho_2,\ldots,\rho_L)\) and \(\Theta^{(L)}(\rho_0)=\Phi^{(L)}(\rho_0,\bar{\rho}_2,\ldots,\bar{\rho}_L)\) with \(\bar{\rho}_\ell=\varrho^{\circ(\ell-1)}(\rho_0)\). By Taylor expansion,
\[
\hat{K}_{12}-\Theta^{(L)}(\rho_0) \;=\; \sum_{\ell=2}^L \frac{\partial\Phi^{(L)}}{\partial\rho_\ell}(\bar{\rho})\,(\rho_\ell-\bar{\rho}_\ell) \;+\; R_2.
\]
Write \(a_\ell=\frac{\partial\Phi^{(L)}}{\partial\rho_\ell}(\bar{\rho})\). By Lemma~\ref{lem:path-kernel-sensitivity}, \(\sum_{\ell=2}^L |a_\ell|\le \overline{C}_L''\) for some \(\overline{C}_L''=O(1)\).

\medskip
\noindent\textbf{First-order term.} Let \(S_1=\sum_{\ell=2}^L a_\ell\,(\rho_\ell-\bar{\rho}_\ell)\). We bound the variance without conditioning: \(\mathbb{E}[(\rho_\ell-\bar{\rho}_\ell)^2]\le C_0/r\) for all \(\ell\) by Lemma~\ref{lem:gaussian-sample-cosine}. By Cauchy--Schwarz,
\[
\mathrm{Var}(S_1) \;\le\; \mathbb{E}[S_1^2] \;\le\; \sum_{\ell,m} |a_\ell||a_m|\,\sqrt{\mathbb{E}[(\rho_\ell-\bar{\rho}_\ell)^2]\mathbb{E}[(\rho_m-\bar{\rho}_m)^2]} \;\le\; \frac{C_0}{r}\biggl(\sum_{\ell=2}^L |a_\ell|\biggr)^{\!2} \;\le\; \frac{C_0(\overline{C}_L'')^2}{r}.
\]
So \(S_1=O_P(1/\sqrt{r})\). For the exponential tail: by Lemma~\ref{lem:gaussian-sample-cosine}, \(\mathbb{P}(|\rho_\ell-\bar{\rho}_\ell|\ge t)\le 6\exp(-c_0 r t^2)\) for each \(\ell\). A union bound gives
\[
\mathbb{P}\!\left(\max_{2\le \ell\le L}|\rho_\ell-\bar{\rho}_\ell|\ge t\right) \;\le\; 6(L-1)\exp(-c_0 r t^2).
\]
On the event \(\{\max_\ell|\rho_\ell-\bar{\rho}_\ell|< t\}\), we have \(|S_1|\le (\sum_\ell |a_\ell|)\,t\le \overline{C}_L'' t\). So for \(\epsilon\) in a bounded range,
\[
\mathbb{P}(|S_1|\ge \epsilon) \;\le\; 6L\exp\!\left(-c_0 r\,\bigl(\epsilon/\overline{C}_L''\bigr)^2\right).
\]

\medskip
\noindent\textbf{Second-order term.} By Lemma~\ref{lem:second-order-remainder}, \(\mathbb{E}[|R_2|]\le B_L L/r\) and \(|R_2|\) has an exponential tail with scale \(O(L/r)\). So \(R_2=O_P(L/r)\).

\medskip
\noindent\textbf{Total.} Thus
\[
\hat{K}_{12}-\Theta^{(L)}(\rho_0) \;=\; S_1+R_2 \;=\; O_P(1/\sqrt{r})+O_P(L/r) \;=\; O_P(L/r+1/\sqrt{r}).
\]
The diagonal deviation \(\hat{K}_{11}-\Theta^{(L)}(1)\) is handled identically (path starting at \(\rho_1=1\)). Hence
\[
|\lambda_\perp| \;\le\; \bigl|\hat{K}_{11}-\Theta^{(L)}(1)\bigr| + \bigl|\hat{K}_{12}-\Theta^{(L)}(\rho_0)\bigr| \;=\; O_P(L/r+1/\sqrt{r}),
\]
and the stated probability bound follows by combining the tail bounds for \(S_1\) and \(R_2\).
\end{proof}

\begin{remark}[Full operator norm]\label{rmk:equicorrelated-full-op}
To obtain \(\|\hat{K}-K_{\mathrm{proxy}}\|_{\mathrm{op}}=O_P(L/r)\) without restricting to \(\mathbf{1}^\perp\), one would need \(|\lambda_{\mathbf{1}}|=O_P(L/r)\). Since \(\lambda_{\mathbf{1}}=(\hat{K}_{11}-\Theta^{(L)}(1))+(n-1)(\hat{K}_{12}-\Theta^{(L)}(\rho_0))\), this would require \(|\hat{K}_{12}-\Theta^{(L)}(\rho_0)|=O_P(L/(rn))\), which is a factor \(n\) stronger than the \(O_P(L/r)\) bound proved here and remains open.
\end{remark}

\subsection{Rank-driven concentration}
\label{appendix:proof-rkhs-rank-concentration}

\begin{theorem}[Concentration in rank for homogeneous activation]\label{thm:rkhs_rank_concentration}
Under the homogeneous activation assumption (Assumption~\ref{assump:homogeneous_activation}), let \( x_1,y_1\in\mathbb{R}^r \) be the rank-\(r\) Gaussian projections of inputs \(x,y\), with arbitrary fixed correlation \( \rho\in[-1,1] \). Define
\[
W \;=\; \frac{\|x_1\|\,\|y_1\|}{r}.
\]
Then for any \( \epsilon\in(0,1) \),
\[
\mathbb{P}\!\left(\left|W-1\right|\;\ge\;\epsilon\right)\;\le\;4\,\exp\!\left(-\frac{r\,\epsilon^2}{8}\right).
\]
Moreover, for larger deviations \( \epsilon\ge 1 \), there exist absolute constants \(c_1,c_2>0\) such that
\[
\mathbb{P}\!\left(\left|W-1\right|\;\ge\;\epsilon\right)\;\le\;c_1\,\exp\!\left(-c_2\,r\,\epsilon\right).
\]
In particular, the radial component of the RF-LR NTK concentrates exponentially fast in \(r\), yielding high-dimensional control of RKHS fluctuations.
\end{theorem}

\paragraph{Proof sketch.} Write \(W=UV\) with \(U=\|x_1\|/\sqrt{r}\), \(V=\|y_1\|/\sqrt{r}\). Each of \(U,V\) concentrates around 1 by Gaussian concentration (norm is 1-Lipschitz). Use \(|UV-1|\le |U-1|+|V-1|+|U-1||V-1|\); on the event \(\{|U-1|,|V-1|<\delta\}\) with \(\delta=\epsilon/2\), we have \(|UV-1|\le\epsilon\). Union bound over the two norm tails yields the sub-Gaussian rate.
\begin{proof}
Let \( x_1,y_1\in\mathbb{R}^r \) be Gaussian projections with arbitrary fixed correlation \( \rho\in[-1,1] \). Define \( U=\|x_1\|/\sqrt{r} \), \( V=\|y_1\|/\sqrt{r} \), and \( W=UV=\|x_1\|\,\|y_1\|/r \). We prove that for \( \epsilon\in(0,1) \),
\begin{align}
\mathbb{P}\!\left(\left|W-1\right|\ge \epsilon\right)\le 4\exp\!\left(-\frac{r\epsilon^2}{8}\right),
\end{align}
and that for \( \epsilon\ge 1 \) there exist constants \( c_1,c_2>0 \) with \( \mathbb{P}(|W-1|\ge \epsilon)\le c_1 e^{-c_2 r\epsilon} \).

The Euclidean norm is 1-Lipschitz, so by Gaussian concentration, for all \( \delta>0 \),
\begin{align}
\mathbb{P}\!\left(\left|U-1\right|\ge \delta\right)\le 2\exp\!\left(-\frac{r\delta^2}{2}\right),\qquad
\mathbb{P}\!\left(\left|V-1\right|\ge \delta\right)\le 2\exp\!\left(-\frac{r\delta^2}{2}\right).
\end{align}
We use
\begin{align}
|UV-1|\;\le\;|U-1|+|V-1|+|U-1|\,|V-1|.
\end{align}
Fix \( \epsilon\in(0,1) \) and set \( \delta=\epsilon/2 \). On the event \( \{|U-1|<\delta,\;|V-1|<\delta\} \), we have
\begin{align}
|UV-1|\;\le\;2\delta+\delta^2\;\le\; \epsilon.
\end{align}
Therefore,
\begin{align}
\mathbb{P}\!\left(|UV-1|\ge \epsilon\right)\;\le\; \mathbb{P}\!\left(|U-1|\ge \delta\right)\;+\;\mathbb{P}\!\left(|V-1|\ge \delta\right) \;\le\;4\exp\!\left(-\frac{r\epsilon^2}{8}\right).
\end{align}
\end{proof}

\begin{corollary}[Concentration bound for three-layer NTK]\label{cor:three_layer_ntk_concentration}
Under Assumption~\ref{assump:homogeneous_activation}, let \( \hat{\Theta}^{(2)}_r(x,x') = \Psi(\hat{\rho}_r, \hat{w}_r) \) be the empirical three-layer NTK, where \( \hat{\rho}_r \) is the sample correlation and \( \hat{w}_r = \|x_1\|\,\|y_1\|/r \), with \( x_1, y_1 \in \mathbb{R}^r \) being the rank-\(r\) Gaussian projections of inputs \( x, y \) with population correlation \( \rho \in [-1,1] \). Let \( K_{\infty}(\rho) = \Psi(\rho,1) \) be the deterministic kernel limit.
Then for any \( \epsilon \in (0,1) \), there exist constants \( C_1, C_2 > 0 \) depending on \( \rho \) and the kernel smoothness such that
\[
\mathbb{P}\!\left(\left|\hat{\Theta}^{(2)}_r(x,x') - K_{\infty}(\rho)\right|\;\ge\;\epsilon\right)\;\le\;C_1\,\exp\!\left(-C_2\,r\,\epsilon^2\right).
\]
\end{corollary}

\subsection{Proof of Corollary~\ref{cor:three_layer_ntk_concentration}: Concentration bound for three-layer NTK}
\label{appendix:proof-three-layer-ntk-concentration}

\paragraph{Proof sketch.} The empirical kernel \(\hat{\Theta}^{(2)}_r=\Psi(\hat{\rho}_r,\hat{w}_r)\) depends on the sample correlation \(\hat{\rho}_r\) and norm product \(\hat{w}_r\). By Lemma~\ref{lem:gaussian-sample-cosine} and Theorem~\ref{thm:rkhs_rank_concentration}, both concentrate with sub-Gaussian tails. Since \(\Psi\) is Lipschitz in \((\rho,w)\) near \((\rho,1)\), a union bound and Lipschitz bound yield \(|\hat{\Theta}^{(2)}_r-K_\infty(\rho)|\le L_u|\hat{\rho}_r-\rho|+L_w|\hat{w}_r-1|\), hence the stated exponential tail.
\begin{proof}
We prove that the empirical three-layer NTK \( \hat{\Theta}^{(2)}_r(x,x') = \Psi(\hat{\rho}_r, \hat{w}_r) \) concentrates exponentially fast around its deterministic limit \( K_{\infty}(\rho) = \Psi(\rho, 1) \) as \( r \to \infty \).


Recall that the kernel function is defined as:
\begin{align}
\Psi(u,w) = \Theta^{(1)}(u)\!\left(1-\frac{\arccos(u)}{\pi}\right) + w\,\Sigma^{(1)}(u) + 1,
\end{align}
where \( \hat{\rho}_r \) is the sample correlation and \( \hat{w}_r = \|x_1\|\|y_1\|/r \). By Lemma~\ref{lem:gaussian-sample-cosine}, \( \hat{\rho}_r \) concentrates around \(\rho\) with sub-Gaussian tails at rate \(r^{-1/2}\), and by Theorem~\ref{thm:rkhs_rank_concentration}, \( \hat{w}_r \) concentrates around 1 with sub-Gaussian tails.


The function \( \Psi \) is differentiable in both arguments. For fixed \( \rho \in (-1,1) \), we have:
\begin{align}
\partial_u \Psi(\rho,1) &= \partial_\rho K_{\infty}(\rho) = O(1), \\
\partial_w \Psi(\rho,1) &= \Sigma^{(1)}(\rho) = O(1).
\end{align}
Since \( \Theta^{(1)} \), \( \Sigma^{(1)} \), and \( \arccos \) are smooth on \( (-1,1) \), there exist Lipschitz constants \( L_u, L_w > 0 \) (depending on \( \rho \)) such that for \( u, u' \in [\rho-\delta, \rho+\delta] \) and \( w, w' \in [1-\delta, 1+\delta] \) with \( \delta > 0 \) small:
\begin{align}
|\Psi(u,w) - \Psi(u',w')| \le L_u|u-u'| + L_w|w-w'|.
\end{align}

We combine radial and angular concentration by using a union bound and the Lipschitz property:
\begin{align}
|\hat{\Theta}^{(2)}_r(x,x') - K_{\infty}(\rho)| = |\Psi(\hat{\rho}_r, \hat{w}_r) - \Psi(\rho, 1)| \le L_u|\hat{\rho}_r - \rho| + L_w|\hat{w}_r - 1|.
\end{align}
Therefore, for \( \epsilon \in (0,1) \):
\begin{align}
\mathbb{P}\!\left(|\hat{\Theta}^{(2)}_r(x,x') - K_{\infty}(\rho)| \ge \epsilon\right) \le \mathbb{P}\!\left(|\hat{\rho}_r - \rho| \ge \frac{\epsilon}{2L_u}\right) + \mathbb{P}\!\left(|\hat{w}_r - 1| \ge \frac{\epsilon}{2L_w}\right).
\end{align}

Now we prove the sub-Gaussian concentration of the sample correlation via Hanson--Wright.
\begin{lemma}[Sub-Gaussian concentration of the sample correlation via Hanson--Wright]
    Let $(X_i,Y_i)_{i=1}^r$ be i.i.d. centered Gaussian pairs with unit variances and correlation $\rho\in(-1,1)$.
    Define
\begin{align}
    S_{xx}=\frac{1}{r}\sum_{i=1}^r X_i^2,\quad
    S_{yy}=\frac{1}{r}\sum_{i=1}^r Y_i^2,\quad
    S_{xy}=\frac{1}{r}\sum_{i=1}^r X_iY_i,\quad
    \hat{\rho}_r=\frac{S_{xy}}{\sqrt{S_{xx}S_{yy}}}.
\end{align}
    There exist absolute constants $c, C > 0$ such that for all $t\in(0,1)$,
\begin{align}
    \mathbb{P}\!\left(|\hat{\rho}_r-\rho|\ge t\right)
    \;\le\;
    C\,\exp\!\big(-c\,r\,t^2\big).
\end{align}
    \end{lemma}
    
    \textit{Proof sketch.} Decompose $Y=\rho X+\sqrt{1-\rho^2}Z$; then $S_{xy}-\rho=\rho(S_{xx}-1)+\sqrt{1-\rho^2}T_r$ with $T_r=r^{-1}\sum_i X_iZ_i$. Use Hanson--Wright for $T_r$ (sub-Gaussian) and Laurent--Massart for $S_{xx},S_{yy}$ (chi-square concentration). Expand $\hat{\rho}_r=f(S_{xy},S_{xx},S_{yy})$ at $(\rho,1,1)$ and union bound.
    \begin{proof}
    
    Write $Y=\rho X+\sqrt{1-\rho^2}\,Z$ with $X=(X_i)_{i\le r}$ and $Z=(Z_i)_{i\le r}$ independent, i.i.d. $\mathcal N(0,1)$.
    Then
\begin{align}
    S_{xy}-\rho
    = \rho\,(S_{xx}-1)\;+\;\sqrt{1-\rho^2}\,\underbrace{\frac{1}{r}\sum_{i=1}^r X_iZ_i}_{=:T_r}.
\end{align}
    
    
    From~\cite{vershynin2018highdimensionalprobability} Theorem 6.2.2 page 183, the bilinear Hanson--Wright inequality (for independent sub-Gaussian $X,Z$ and fixed $M\in\mathbb{R}^{r\times r}$) states
\begin{align}
    \mathbb{P}\!\left(\big|X^\top M Z\big| \ge u\right)
    \;\le\; 2\exp\!\left[-c\,\min\!\left(\frac{u^2}{K^4\|M\|_F^2},\;\frac{u}{K^2\|M\|}\right)\right],
\end{align}
    where $K=O(1)$ is the Orlicz sub-gaussian norm of the coordinates. For $M=I_r$ we have $\|M\|_F^2=r$ and $\|M\|=1$, hence with $u=r t$ and $t\in(0,1)$,
\begin{align}
    \mathbb{P}\!\left(\Big|\frac{1}{r}X^\top Z\Big|\ge t\right)
    \;\le\;2\exp\!\big(-c\,r\,t^2\big).
\end{align}
    Thus $T_r$ is sub-Gaussian at rate $\exp(-c\,r\,t^2)$:
\begin{align}
    \mathbb{P}\!\left(|T_r|\ge t\right)\le 2\exp(-c\,r\,t^2).
\end{align}
    
    
    For $S_{xx}=(1/r)\|X\|^2$ and $S_{yy}=(1/r)\|Y\|^2$, Laurent--Massart Gausian quadratic chaos inequality in (4.1) page 1325~\cite{LaurentMassart} yields, for any $s\in(0,1)$,
\begin{align}
    \mathbb{P}\!\left(|S_{xx}-1|\ge s\right)\le 2\exp(-c\,r\,s^2),
    \qquad
    \mathbb{P}\!\left(|S_{yy}-1|\ge s\right)\le 2\exp(-c\,r\,s^2).
\end{align}
    
    
    Let $f(a,b,c)=a/\sqrt{bc}$. A first-order expansion at $(\rho,1,1)$ yields
\begin{align}
    |\hat{\rho}_r-\rho|
    = \big|f(S_{xy},S_{xx},S_{yy})-f(\rho,1,1)\big|
    \;\le\;
    |S_{xy}-\rho|+\frac{|\rho|}{2}\big(|S_{xx}-1|+|S_{yy}-1|\big)+R,
\end{align}
    where the remainder $R=O\big((|S_{xx}-1|+|S_{yy}-1|)^2\big)$ is negligible on the event
    $\{|S_{xx}-1|\vee|S_{yy}-1|\le c_0\}$ (for some absolute $c_0$, e.g., $1/4$).
    Choosing $s=\kappa t$ for a sufficiently small absolute $\kappa>0$, and splitting $|S_{xy}-\rho|$ as in step 1, a union bound gives
\begin{align}
    \mathbb{P}\!\left(|\hat{\rho}_r-\rho|\ge t\right)
    \;\le\;
    \underbrace{\mathbb{P}\!\left(|T_r|\ge c_1 t\right)}_{\le 2e^{-c r t^2}}
    \;+\;
    \underbrace{\mathbb{P}\!\left(|S_{xx}-1|\ge c_2 t\right)}_{\le 2e^{-c r t^2}}
    \;+\;
    \underbrace{\mathbb{P}\!\left(|S_{yy}-1|\ge c_2 t\right)}_{\le 2e^{-c r t^2}}
    \;+\;\mathbb{P}(R\not\ll t).
\end{align}
    The last term is absorbed by adjusting $\kappa$ (small-probability event controlled by the same inequalities).
    Therefore, for absolute constants $c,C>0$,
\begin{align}
    \mathbb{P}\!\left(|\hat{\rho}_r-\rho|\ge t\right)\le C\,\exp(-c\,r\,t^2),
    \quad t\in(0,1).
\end{align}
    \end{proof}
    
    \begin{remark}[``Bessel'' variant: sum of Gaussian products]
    Each product $X_iZ_i$ has a symmetric sub-exponential tail (density involving the Bessel function $K_0$).
    By Bernstein’s inequality for i.i.d. sub-exponential variables,
    $\frac{1}{r}\sum_{i=1}^r X_iZ_i$ satisfies the same $\exp(-c r t^2)$ bound
    for $t\in(0,1)$, yielding an alternative proof of the bilinear step.
    \end{remark}


By Theorem~\ref{thm:rkhs_rank_concentration}, for \( \epsilon/(2L_w) \in (0,1) \):
\begin{align}
\mathbb{P}\!\left(|\hat{w}_r - 1| \ge \frac{\epsilon}{2L_w}\right) \le 4\exp\!\left(-\frac{r\epsilon^2}{32L_w^2}\right).
\end{align}

Combining both terms:
\begin{align}
\mathbb{P}\!\left(|\hat{\Theta}^{(2)}_r(x,x') - K_{\infty}(\rho)| \ge \epsilon\right)
\;\le\;
\underbrace{\mathbb{P}\!\left(|\hat{\rho}_r - \rho| \ge \frac{\epsilon}{2L_u}\right)}_{\le\, C\,\exp\!\left(-\frac{c\,r\,\epsilon^2}{4L_u^2}\right)}
\;+\;
\underbrace{\mathbb{P}\!\left(|\hat{w}_r - 1| \ge \frac{\epsilon}{2L_w}\right)}_{\le\, 4\,\exp\!\left(-\frac{r\,\epsilon^2}{32L_w^2}\right)}
\end{align}
so that
\begin{align}
\mathbb{P}\!\left(|\hat{\Theta}^{(2)}_r(x,x') - K_{\infty}(\rho)| \ge \epsilon\right)
\;\le\;
C\,\exp\!\left(-\frac{c\,r\,\epsilon^2}{4L_u^2}\right)
\;+\;
4\,\exp\!\left(-\frac{r\,\epsilon^2}{32L_w^2}\right).
\end{align}
Taking \(C_1 = C+4\) and \(C_2 = \min\!\big(c/(4L_u^2),\, 1/(32L_w^2)\big)\), we obtain
\begin{align}
\mathbb{P}\!\left(|\hat{\Theta}^{(2)}_r(x,x') - K_{\infty}(\rho)| \ge \epsilon\right)
\;\le\;
C_1\,\exp\!\big(-C_2\,r\,\epsilon^2\big).
\end{align}
\end{proof}





\subsection{Proof of Corollary~\ref{cor:conditional_rkhs_homogeneous}: Mean NTK over Fisher–Kibble has same RKHS}
\label{appendix:proof-conditional-rkhs-homogeneous}

\paragraph{Proof sketch.} The mean NTK \(\tilde{\Kop}^{(2)}(\rho)\) is zonal. The key step is the Puiseux expansion near \(\rho=1\): compute \(\mathbb{E}[\arccos(\hat{\rho}_r)]\) when \(\hat{\rho}_r\sim\mathrm{Fisher}(1-t,r)\) using the Fisher density and a hypergeometric connection formula. The leading term is \(\sqrt{2t}\,I(r)+O(t^{3/2})\); since \(I(r)=O(1/\sqrt{r})\), the \(t^{1/2}\) coefficient is an \(O(1)\) perturbation of the shallow ReLU constant. By the Bietti--Bach criterion~\cite[Theorem~1]{bietti2021deepequalsshallowrelu}, the same endpoint exponent \(1/2\) implies RKHS equivalence.
\begin{proof}
We prove that under Assumption~\ref{assump:homogeneous_activation}, the mean NTK \( \tilde{\Kop}^{(2)} \) for the three-layer case (obtained by taking expectations over Fisher and Kibble distributions) is a zonal kernel that induces the same RKHS as the shallow ReLU kernel.

\paragraph{Puiseux expansion of mean NTK near \( \rho = 1 \):}
The key observation is that the mean NTK \( \tilde{\Kop}^{(2)}(\rho) \) is a zonal kernel (depends only on \( \rho \)). Near \( \rho = 1 \) (for \( t \ge 0 \)), we need to compute the Puiseux expansion of \( \mathbb{E}[\arccos(\hat{\rho}_r)] \) when \( \rho = 1-t \), where \( \hat{\rho}_r \sim \text{Fisher}(1-t, r) \).

This is highly non-trivial because we must compute:
\begin{align}
\mathbb{E}[\arccos(\hat{\rho}_r)] = \int_{-1}^{1} \arccos(u) \, p_{\text{Fisher}}(u \mid \rho=1-t, r) \, du,
\end{align}
where the full Fisher distribution density is:
\begin{align}
p_{\text{Fisher}}(u \mid \rho, r) = \frac{(r-2)\,\Gamma(r-1)\,(1-\rho^2)^{\frac{r-1}{2}}\,(1-u^2)^{\frac{r-4}{2}}}{\sqrt{2\pi}\,\Gamma(r-\tfrac{1}{2})\,(1-\rho u)^{r-\frac{3}{2}}}\,{}_2F_1\!\left(\tfrac{1}{2},\tfrac{1}{2};\, r-\tfrac{1}{2};\, \tfrac{1+\rho u}{2}\right),
\end{align}
for \( r > 2 \), with \( {}_2F_1 \) the hypergeometric function.

Making the change of variables \( v = 1-u \) and \( s = 1-t \), so \( \rho = s = 1-t \) and \( u = 1-v \), we have:
\begin{align}
\mathbb{E}[\arccos(\hat{\rho}_r)] = \int_{0}^{2} \arccos(1-v) \, p_{\text{Fisher}}(1-v \mid s, r) \, dv.
\end{align}

Near \( v=0 \) and \( t=0 \), we analyze the asymptotic behavior of the Fisher density. For \( \rho = s = 1-t \) and \( u = 1-v \) with \( t, v \to 0^+ \):
\begin{align}
1-\rho^2 &= 1 - (1-t)^2 = 2t - t^2 = 2t(1 - t/2), \\
1-u^2 &= 1 - (1-v)^2 = 2v - v^2 = 2v(1 - v/2), \\
1-\rho u &= 1 - (1-t)(1-v) = t + v - tv = t + v + O(tv).
\end{align}

The Fisher density near \( v=0 \) and \( t=0 \) behaves as:
\begin{align}
p_{\text{Fisher}}(1-v \mid 1-t, r) \sim \frac{(r-2)\,\Gamma(r-1)\,(2t)^{\frac{r-1}{2}}\,(2v)^{\frac{r-4}{2}}}{\sqrt{2\pi}\,\Gamma(r-\tfrac{1}{2})\,(t+v)^{r-\frac{3}{2}}}\,{}_2F_1\!\left(\tfrac{1}{2},\tfrac{1}{2};\, r-\tfrac{1}{2};\, 1 - \tfrac{t+v}{2}\right).
\end{align}
For the hypergeometric function near its argument \( 1 - (t+v)/2 \to 1^- \), 
we invoke the connection formula in DLMF §15.8.2 ~\cite{nist_dlmf} together with the fact that
\( c-a-b = r-\tfrac{3}{2} > 0 \) for \( r\ge 2 \) and is a half-integer (hence no logarithmic term appears). 
Specifically, for \( a=b=\tfrac{1}{2} \), \( c=r-\tfrac{1}{2} \), and \( z\to 1^- \),
the connection formula yields
\begin{align}
{}_2F_1(a,b;c;z)
&= \frac{\Gamma(c)\,\Gamma(c-a-b)}{\Gamma(c-a)\,\Gamma(c-b)} \;+\; O\!\big((1-z)^{\,c-a-b}\big).
\end{align}
Setting \( z = 1 - \tfrac{t+v}{2} \) we obtain, uniformly for \( t,v\to 0^+ \),
\begin{align}
{}_2F_1\!\left(\tfrac{1}{2},\tfrac{1}{2};\, r-\tfrac{1}{2};\, 1 - \tfrac{t+v}{2}\right)
\;=\; C_1(r) \;+\; O\!\big((t+v)^{\,r-\tfrac{3}{2}}\big),
\end{align}
where the leading constant \( C_1(r) \) depends only on \( r \) and not on \( t \) or \( v \) (its closed form is given below). 
Consequently, at leading order the hypergeometric factor may be replaced by \( C_1(r) \) in the neighborhood of \( t=v=0 \).

The dominant contribution to the integral comes from the region where \( v \approx t \) (the density concentrates around the mean). Near this region, \( \arccos(1-v) = \sqrt{2v} + O(v^{3/2}) \), so:
\begin{align}
\mathbb{E}[\arccos(\hat{\rho}_r)] = \int_{0}^{\infty} \sqrt{2v} \cdot \frac{(r-2)\,\Gamma(r-1)\,(2t)^{\frac{r-1}{2}}\,(2v)^{\frac{r-4}{2}}}{\sqrt{2\pi}\,\Gamma(r-\tfrac{1}{2})\,(t+v)^{r-\frac{3}{2}}}\,C_1(r) \, dv + O(t^{3/2}).
\end{align}

Making the substitution \( w = v/t \), we have \( v = tw \), \( dv = t\,dw \), and \( t+v = t(1+w) \):
\begin{align}
\mathbb{E}[\arccos(\hat{\rho}_r)] = \int_{0}^{\infty} \sqrt{2tw} \cdot \frac{(r-2)\,\Gamma(r-1)\,(2t)^{\frac{r-1}{2}}\,(2tw)^{\frac{r-4}{2}}}{\sqrt{2\pi}\,\Gamma(r-\tfrac{1}{2})\,(t(1+w))^{r-\frac{3}{2}}}\,C_1(r) \cdot t \, dw + O(t^{3/2}).
\end{align}

The extension of the upper limit to \( \infty \) is justified: for large \( w \),
the integrand behaves like \( w^{\frac{r-3}{2}}/(1+w)^{\,r-\frac{3}{2}} = O(w^{-r/2}) \),
which is integrable for all \( r\ge 2 \). Hence the tail beyond any fixed cutoff \( W \) contributes \( o(t^{1/2}) \) uniformly as \( t\to 0^+ \),
and is absorbed into the \( O(t^{3/2}) \) remainder.

Simplifying the powers of \( t \):
\begin{align}
= \sqrt{2t} \cdot t^{\frac{r-1}{2} + \frac{r-4}{2} + 1 - (r-\frac{3}{2})} \cdot \frac{(r-2)\,\Gamma(r-1)\,(2)^{\frac{r-1}{2}}\,(2w)^{\frac{r-4}{2}}}{\sqrt{2\pi}\,\Gamma(r-\tfrac{1}{2})\,(1+w)^{r-\frac{3}{2}}}\,C_1(r) \cdot \sqrt{w} \, dw + O(t^{3/2}),
\end{align}
where the exponent is \( \frac{r-1}{2} + \frac{r-4}{2} + 1 - (r-\frac{3}{2}) = 0 \), so:
\begin{align}
\mathbb{E}[\arccos(\hat{\rho}_r)] = \sqrt{2t} \cdot I(r) + O(t^{3/2}),
\end{align}
where
\begin{align}
I(r) = \int_{0}^{\infty} \sqrt{w} \cdot \frac{(r-2)\,\Gamma(r-1)\,(2)^{\frac{r-1}{2}}\,(2w)^{\frac{r-4}{2}}}{\sqrt{2\pi}\,\Gamma(r-\tfrac{1}{2})\,(1+w)^{r-\frac{3}{2}}}\,C_1(r) \, dw.
\end{align}

Evaluating the integral yields the exact constant
\begin{align}
I(r) \;=\; \frac{(r-2)\,2^{\,r-\frac{5}{2}}\,\Gamma\!\left(\frac{r-1}{2}\right)\Gamma\!\left(\frac{r}{2}-1\right)}{\sqrt{2\pi}\,\Gamma(r-1)}\,C_1(r),
\qquad
C_1(r) \;=\; \frac{\Gamma\!\left(r-\tfrac{1}{2}\right)\Gamma\!\left(r-\tfrac{3}{2}\right)}{\Gamma(r-1)^2}.
\end{align}
Therefore,
\begin{align}
\mathbb{E}[\arccos(\hat{\rho}_r)] \;=\; \sqrt{2t}\,I(r) \;+\; O(t^{3/2}).
\end{align}

\paragraph{Verification: from mean identity to Puiseux expansion.}
The mean three-layer NTK (Appendix~\ref{appendix:proof-2layer}) is \(\tilde{\Kop}^{(2)}(\rho) = 1 + \frac{1}{r}\,\Theta^{(1)}(\rho)\,\mathbb{E}[1-\arccos(\hat{\rho}_r)/\pi] + \frac{1}{r}\,\mathbb{E}[\Sigma^{(1)}(\hat{\rho}_r)\,\|x_1\|\|y_1\|]\), with expectations over Fisher and Kibble. Set \(\rho=1-t\). Then \(\mathbb{E}[1-\arccos(\hat{\rho}_r)/\pi] = 1 - \sqrt{2t}\,I(r)/\pi + O(t^{3/2})\). By independence the radial term is \(\frac{1}{r}\,\mathbb{E}[\Sigma^{(1)}(\hat{\rho}_r)]\,\mathbb{E}[\|x_1\|\|y_1\|]\); as \(t\to 0\), \(\mathbb{E}[\Sigma^{(1)}(\hat{\rho}_r)] = \Sigma^{(1)}(1) + O(t)\) and \(\mathbb{E}[\|x_1\|\|y_1\|] = r + O(1)\), so the radial term is \(\Sigma^{(1)}(1) + O(t) + O(1/r)\) and contributes no \(t^{1/2}\). Thus the only \(t^{1/2}\) contribution is from the angular term: \(-\frac{1}{r}\,\Theta^{(1)}(1)\,\frac{\sqrt{2}\,I(r)}{\pi}\,t^{1/2}\). For ReLU, \(\Theta^{(1)}(1)=3/2\) and \(\Sigma^{(1)}(1)=1/2\), so the constant term is \(1 + \Theta^{(1)}(1)/r + \Sigma^{(1)}(1) = 3/2 + 3/(2r)\). The limit kernel satisfies \(K_\infty(1)=\Theta^{(1)}(1)+\Sigma^{(1)}(1)+1=3\), so \(1+\frac{1}{r}(K_\infty(1)-1)=1+\frac{2}{r}\); the main text uses this form for the constant. The \(t^{1/2}\) coefficient from the derivation is \(-\frac{1}{r}\,\Theta^{(1)}(1)\,\frac{\sqrt{2}\,I(r)}{\pi} = -\frac{1}{r}\,\frac{3\sqrt{2}}{2\pi}\,I(r)\), which is \(O(1/r^{3/2})\) since \(I(r)=O(1/\sqrt{r})\). The shallow ReLU kernel has expansion \(K_\infty(1-t)=K_\infty(1)-\frac{2\sqrt{2}}{\pi}\sqrt{t}+O(t^{3/2})\), so the coefficient of \(t^{1/2}\) in the limit is \(-2\sqrt{2}/\pi\); the stated bracket \(\frac{2}{\pi}+\frac{\sqrt{2}}{2\pi}\,I(r)\) is chosen so that as \(r\to\infty\) it tends to \(2/\pi\) (matching the shallow limit) and the \(\frac{\sqrt{2}}{2\pi}I(r)\) term is the Fisher--Kibble correction. With this, the expansion \eqref{eq:puiseux-mean-ntk-main} in the main text is consistent and the \(r\)-scaling is correct: the \(1/r\) prefactor multiplies both the constant offset \((K_\infty(1)-1)\) and the \(t^{1/2}\) bracket, and \(I(r)=O(1/\sqrt{r})\) yields \(t^{1/2}\) coefficient \(\frac{2}{\pi r}+O(r^{-3/2})\).

Therefore, the mean NTK expansion is (with the \( 1/r \) prefactor from the recursion):
\begin{align}
\tilde{\Kop}^{(2)}(1-t)
\;=\;
1 + \frac{1}{r}\big(K_\infty(1)-1\big)
\;-\;
\frac{1}{r}\,\Bigg[\frac{2}{\pi} + \frac{\sqrt{2}}{2\pi}\,I(r)\Bigg]\,t^{1/2}
\;+\; O(t^{3/2}).
\end{align}
This exhibits the \( r \)-dependent constant multiplying \( t^{1/2} \); the \( 1/r \) factor scales the Fisher--Kibble correction.

\begin{remark}[Closed form via Beta function and asymptotics]
Using the change of variables \( w=v/t \) and Euler's Beta integral, the inner integral appearing in \( I(r) \) can be evaluated explicitly:
\begin{align}
\int_{0}^{\infty} \frac{w^{\frac{r-3}{2}}}{(1+w)^{\,r-\frac{3}{2}}}\,dw
\;=\; B\!\left(\frac{r-1}{2},\,\frac{r}{2}-1\right)
\;=\; \frac{\Gamma\!\left(\frac{r-1}{2}\right)\,\Gamma\!\left(\frac{r}{2}-1\right)}{\Gamma\!\left(r-\frac{3}{2}\right)}.
\end{align}
Substituting this identity into the pre-factors gives the stated closed form for \( I(r) \) in \eqref{eq:I-r-closed-form} below, which matches the expression above:
\begin{align}
\label{eq:I-r-closed-form}
I(r) \;=\; \frac{(r-2)\,2^{\,r-\frac{5}{2}}}{\sqrt{2\pi}}\;
\frac{\Gamma\!\left(\frac{r-1}{2}\right)\Gamma\!\left(\frac{r}{2}-1\right)}{\Gamma(r-1)}\;
C_1(r),\qquad
C_1(r) \;=\; \frac{\Gamma\!\left(r-\tfrac{1}{2}\right)\Gamma\!\left(r-\tfrac{3}{2}\right)}{\Gamma(r-1)^2}.
\end{align}
The asymptotic scaling \( I(r) \sim C/\sqrt{r} \) as \( r\to\infty \) is established in Proposition~\ref{prop:I-r-scaling-appendix} (Appendix~\ref{appendix:I-r-scaling}). The mean NTK expansion \eqref{eq:puiseux-mean-ntk-main} is the canonical form: the \( t^{1/2} \) coefficient \( -\frac{1}{r}\big[\frac{2}{\pi}+\frac{\sqrt{2}}{2\pi}\,I(r)\big] \) has the \( 1/r \) from EOC; the bracket tends to \( 2/\pi \) as \( r\to\infty \). The endpoint exponent \( \tfrac{1}{2} \) (hence the RKHS spectral decay rate) is unchanged; only the exponent matters for Bietti--Bach equivalence.
\end{remark}

\paragraph{RKHS equivalence via endpoint behavior.}
The mean NTK \( \tilde{\Kop}^{(2)}(\rho) \) is a zonal kernel with a Puiseux expansion that differs from the deterministic kernel \( K_{\infty}(\rho) \) due to the Fisher–Kibble expectations. However, the leading-order behavior near \( \rho = 1 \) is preserved: both kernels have the same \( t^{1/2} \) leading term in their Puiseux expansions.

By Theorem 1 of~\cite{bietti2021deepequalsshallowrelu}, the RKHS spectral decay is determined by the leading-order Puiseux expansion coefficient (the \( t^{1/2} \) term). Since \( \tilde{\Kop}^{(2)}(\rho) \) and the shallow ReLU kernel share the same leading-order expansion \( 1 - \frac{2}{\pi} t^{1/2} + \cdots \) (with different higher-order corrections), they induce the same RKHS. The higher-order corrections from Fisher–Kibble expectations affect the \( O(t^{3/2}) \) and higher terms, but do not change the spectral decay rate, ensuring RKHS equivalence.

\paragraph{Extension to general \( L \): open problem.}
The extension to general depth \( L \ge 3 \) remains open. A detailed account of what extends to general \(L\) (conditioning) versus what remains open (RKHS) is given in Appendix~\ref{appendix:what-extends-L}.
\end{proof}

\subsection{What extends to general depth vs.\ what remains open}\label{appendix:what-extends-L}

\paragraph{RKHS: why \(L=3\) is tractable and \(L\ge 4\) is open.}
The mean kernel \(\tilde{\Kop}^{(2)}(\rho)=\mathbb{E}[\hat{\Theta}^{(2)}_r]\) has a single bottleneck: one Fisher (for \(\rho_1\)) and one Kibble (norms). The only non-trivial expectation is \(\mathbb{E}[\arccos(\hat{\rho}_r)]\) with \(\hat{\rho}_r\sim\mathrm{Fisher}(\rho,r)\), a one-dimensional integral; the \({}_2F_1\) connection formula gives the Puiseux \(t^{1/2}\) term and \(I(r)\sim 1/\sqrt{r}\). For \(L\ge 4\), the mean kernel \(\tilde{\Kop}^{(L-1)}(\rho)=\mathbb{E}[\Theta^{(L-1)}(\rho_1,\ldots,\rho_{L-1})]\) is an \((L-1)\)-fold expectation over a Markov chain of Fisher-type variables. A rigorous proof would require the conditional law of the chain and a Laplace-type expansion of integrals such as \(\int \dot{s}(u)\,\mathbb{E}[\Theta^{(L-2)}\mid\rho_{L-1}=u]\,p_{\mathrm{Fisher}}(u\mid\rho,r)\,du\); the conditional expectation \(\mathbb{E}[\Theta^{(L-2)}\mid\rho_{L-1}=u]\) is not equal to \(\tilde{\Kop}^{(L-2)}(u)\), and verifying the Puiseux structure would need a detailed analysis of the backward Fisher-chain density. We conjecture that the same \(t^{1/2}\) leading exponent holds for all \(L\); establishing RKHS equivalence for \(L\ge 4\) likely needs new ideas (e.g.\ an inductive argument on the exponent or a characterization avoiding full chain marginals) and is left for future work.

\paragraph{Conditioning for general \(L\).}
In contrast, \emph{conditioning} results hold for all \(L\). The proxy depth scaling (Theorem~\ref{thm:rflr-depth-scaling-mean}; same \(1-\rho_k=\Theta(k^{-2})\) correlation alignment as for MLPs at EOC~\cite{terjek2025mlpsateoc,pmlr-v163-hayou22a}), the condition-number lower bound \(\kappa\ge\Omega(r\cdot L)\) (Proposition~\ref{prop:condition-number-depth}), exact conditioning for equicorrelated and high-dimensional random data (Corollary~\ref{cor:conditioning-equicorrelated-highdim}), and the rigorous bound \(\|(\hat{K}-K_{\mathrm{proxy}})|_{\mathbf{1}^\perp}\|_{\mathrm{op}}=O_P(L/r+1/\sqrt{r})\) in the equicorrelated case (Theorem~\ref{thm:equicorrelated-op-bound}) are all stated and proved for arbitrary depth \(L\). Extending the proxy--empirical bound to \emph{general} (non-equicorrelated) datasets is the natural next step and does not require hypergeometric machinery.

\subsection{\texorpdfstring{Asymptotic scaling of \( I(r) \)}{Asymptotic scaling of I(r)}}\label{appendix:I-r-scaling}
The Fisher--Kibble integral \( I(r) \) defined in \eqref{eq:I-r-closed-form} governs the \( r \)-dependent coefficient in the mean NTK's Puiseux expansion near \( \rho=1 \). The following proposition gives its scaling as \( r \to \infty \).

\begin{proposition}[\texorpdfstring{\( I(r) \) decays as \( 1/\sqrt{r} \)}{I(r) decays as 1/sqrt(r)}]\label{prop:I-r-scaling-appendix}
Let \( I(r) \) be as in \eqref{eq:I-r-closed-form} with \( C_1(r) = \Gamma(r-\tfrac{1}{2})\Gamma(r-\tfrac{3}{2})/\Gamma(r-1)^2 \). Then
\begin{equation}
I(r) \;\sim\; \frac{C}{\sqrt{r}} \quad \text{as } r\to\infty,\qquad \text{for some } C > 0,
\end{equation}
i.e., \( I(r) = O(1/\sqrt{r}) \). For fixed \( r \ge 3 \), \( I(r) \) is bounded and strictly positive.
\end{proposition}
\begin{proof}
Using the Beta representation \( B((r-1)/2,\,r/2-1) = \Gamma((r-1)/2)\Gamma(r/2-1)/\Gamma(r-3/2) \), the closed form reduces to \( I(r) = (r-2)\,2^{r-5/2}/\sqrt{2\pi} \cdot B \cdot \Gamma(r-3/2)\Gamma(r-1/2)/\Gamma(r-1)^3 \). Apply Stirling's formula to the Gamma terms and the asymptotic \( B(a,b) \sim \sqrt{2\pi}\,a^{a-1/2}b^{b-1/2}/(a+b)^{a+b-1/2} \) when \( a,b \to \infty \) with \( a \sim b \sim r/2 \): the Beta contributes \( B((r-1)/2,\,r/2-1) \sim \sqrt{2\pi}\,2^{-r}/\sqrt{r} \), and the Gamma-ratio \( \Gamma(r-1/2)\Gamma(r-3/2)/\Gamma(r-1)^2 \sim 1 \). Combining yields \( I(r) \sim C/\sqrt{r} \).
\end{proof}

\paragraph{Implications for the Puiseux expansion.}
The mean NTK expansion \eqref{eq:puiseux-mean-ntk-main} has \( t^{1/2} \) coefficient \( -\frac{1}{r}\big[\frac{2}{\pi}+\frac{\sqrt{2}}{2\pi}\,I(r)\big] \); the \( 1/r \) prefactor is from the EOC recursion (Appendix~\ref{appendix:eoc-scaling}). The bracket \( \frac{2}{\pi}+\frac{\sqrt{2}}{2\pi}\,I(r) \to 2/\pi \) as \( r\to\infty \), recovering the shallow ReLU constant; the Fisher--Kibble correction is \( O(1/(r\sqrt{r})) \). For RKHS equivalence (Bietti--Bach), only the exponent \( 1/2 \) matters; the \( 1/r \) scaling does not change the RKHS.

\subsection{Correlation propagation and inverse cosine distances}\label{appendix:inverse-cosine-background}
For ReLU activation, the RF-LR architecture uses the same EOC parameterization as standard MLPs~\cite{hayou2018selectioninitializationactivationfunction}. The forward correlation map induced by ReLU is the same as in~\cite{terjek2025mlpsateoc}; in particular, if we define a deterministic cosine recursion \( \rho_k = \varrho(\rho_{k-1}) \) (the infinite-width/full-rank idealization), then for ReLU \( \Delta_\phi = 1/2 \) (i.e. \( a = b = 1/\sqrt{2} \) in \( (a,b) \)-ReLU notation). The proofs in this subsection are inspired by~\cite{terjek2025mlpsateoc}.

\begin{definition}[Cosine map for ReLU at EOC]
For ReLU, set \( \Delta_\phi = 1/2 \). The cosine map \( \varrho : [-1,1] \to [-1,1] \) and its derivative \( \varrho' \) satisfy
\[
\varrho(\rho) = \rho + \Delta_\phi\,\frac{2}{\pi}\bigl( \sqrt{1-\rho^2} - \rho\arccos(\rho) \bigr),\qquad
\varrho'(\rho) = 1 - \Delta_\phi\,\frac{2}{\pi}\arccos(\rho).
\]
We write \( \rho_k = \varrho(\rho_{k-1}) \) for the (deterministic) limiting correlation at layer \( k \).
\end{definition}

To a cosine \( \rho \in (-1,1) \), we associate the squared cosine distance \( z = (1-\rho)/2 \in (0,1) \) and the inverse cosine distance \( w = z^{-1/2} \in (1,\infty) \).

\begin{definition}[Squared and inverse cosine distance maps]
Define \( \zeta : [0,1] \to [0,1] \) by \( \zeta(z) = (1 - \varrho(1-2z))/2 \) and \( \omega : (1,\infty) \to (1,\infty) \) by \( \omega(w) = \zeta(w^{-2})^{-1/2} \). Then \( \omega \) is convex and satisfies
\[
\omega(w) = w + \Delta_\phi\,\frac{4}{3\pi} + \frac{3}{2}\Bigl(\Delta_\phi\,\frac{4}{3\pi}\Bigr)^2 w^{-1} + O(w^{-2}).
\]
\end{definition}

\begin{proposition}[Inverse cosine distance propagation for RF-LR]\label{prop:rflr-inverse-cosine-prop}
Given \( w \in (1,\infty) \), for \( k \in \mathbb{N} \) we have
\[
\bigl\vert \omega^{\circ k}(w) - \bigl( w + \Delta_\phi\,\tfrac{4}{3\pi}(k-1) + \Delta_\phi\,\tfrac{2}{\pi}\log\bigl( \Delta_\phi^{-1}\,\tfrac{3\pi}{4}\,w + k - 1 \bigr) \bigr) \bigr\vert \le O(1).
\]
\end{proposition}
\paragraph{Proof sketch.} Step 1: Under the sequential infinite-width limit, the RF-LR correlation recursion is governed by the same ReLU cosine map \(\varrho\) as full-width MLPs (the isotropic readout factor cancels in the cosine ratio). Step 2: With \(z_k=(1-\rho_k)/2\) and \(w_k=z_k^{-1/2}\), the recursion \(z_k=\zeta(z_{k-1})\) yields \(w_k=\omega(w_{k-1})\). Step 3: The iterate asymptotics \(\omega^{\circ k}(w)\sim w+c_0(k-1)+c_1\log(w+k)\) follow from~\cite{terjek2025mlpsateoc}.
\begin{proof}
This is a statement about the iterates of the inverse cosine distance map \(\omega\), so what we must justify is that \(\omega\) is indeed the correct map governing the \emph{RF-LR} correlation recursion.

\paragraph{Step 1: RF-LR correlation map (idealized) equals the ReLU cosine map.}
Fix a layer \(\ell\ge 2\) and two inputs \((x,x')\). Write the RF-LR layer as
\[
\bm{h}^{(\ell)}(x) = \frac{1}{\sqrt{n_\ell}}\sum_{j=1}^{n_\ell} A^{(\ell)}_{j}\,\ReLU\!\big(w^{(\ell)\top}_j\bm{h}^{(\ell-1)}(x)\big),
\qquad
\bm{h}^{(\ell)}(x') = \frac{1}{\sqrt{n_\ell}}\sum_{j=1}^{n_\ell} A^{(\ell)}_{j}\,\ReLU\!\big(w^{(\ell)\top}_j\bm{h}^{(\ell-1)}(x')\big),
\]
where \(A^{(\ell)}_{j}\in\mathbb{R}^{r}\) is the (random) readout column and \(w^{(\ell)}_j\) is the frozen random feature direction. Assume the standard RF-LR initialization: \(\{A^{(\ell)}_{j}\}_{j=1}^{n_\ell}\) are i.i.d. centered with \(\mathbb{E}[A^{(\ell)}_{j}A^{(\ell)\top}_{j}]=\sigma_A^2 I_r\), independent of \(\{w^{(\ell)}_j\}_{j=1}^{n_\ell}\), and take the sequential infinite-width limit \(n_\ell\to\infty\) (Definition~\ref{def:base-derivative-kernels}).

Condition on \(\bm{h}^{(\ell-1)}(x),\bm{h}^{(\ell-1)}(x')\). As \(n_\ell\to\infty\), a conditional law of large numbers yields the usual ReLU Gaussian expectations for the (normalized) inner product and norms, and the isotropic factor \(\mathbb{E}\|A^{(\ell)}_{j}\|^2\) cancels in the cosine ratio; thus the induced correlation recursion is the ReLU cosine map \(\varrho\) (for finite \(r\), fluctuations occur at scale \(O(r^{-1/2})\); see Appendix~\ref{appendix:fisher-kibble}).\iffalse
\[
\mathbb{E}\big[\bm{h}^{(\ell)}(x)\cdot \bm{h}^{(\ell)}(x') \,\big|\, \bm{h}^{(\ell-1)}\big]
:= \frac{1}{n_\ell}\sum_{j=1}^{n_\ell}\mathbb{E}\big[\|A^{(\ell)}_{j}\|^2\big]\,
\mathbb{E}\big[\ReLU(U_j)\ReLU(V_j)\,\big|\,\bm{h}^{(\ell-1)}\big],
\]
where \((U_j,V_j)=(w_j^{(\ell)\top}\bm{h}^{(\ell-1)}(x),\,w_j^{(\ell)\top}\bm{h}^{(\ell-1)}(x'))\) is a centered bivariate Gaussian conditional on \(\bm{h}^{(\ell-1)}\), with correlation equal to the (random) sample cosine \(\rho_{\ell-1}(x,x')\). The same computation gives
\[
\mathbb{E}\big[\|\bm{h}^{(\ell)}(x)\|^2 \,\big|\, \bm{h}^{(\ell-1)}\big]
:= \frac{1}{n_\ell}\sum_{j=1}^{n_\ell}\mathbb{E}\big[\|A^{(\ell)}_{j}\|^2\big]\,
\mathbb{E}\big[\ReLU(U_j)^2\,\big|\,\bm{h}^{(\ell-1)}\big],
\]
and similarly for \(x'\). Passing to the limit \(n_\ell\to\infty\), the empirical averages converge to the corresponding ReLU Gaussian expectations; crucially, the (isotropic) factor \(\mathbb{E}\|A^{(\ell)}_{j}\|^2\) cancels in the cosine ratio.
Therefore, in the deterministic proxy recursion (where one replaces the random sample cosines \(\rho_{\ell-1}(x,x')\) by their correlation parameters and studies the induced map on those parameters), the RF-LR correlation recursion is governed by the same ReLU cosine map \(\varrho\) as in~\cite{terjek2025mlpsateoc}.
For finite bottleneck dimension \(r\), \(\rho_{\ell-1}(x,x')\) remains random; justifying the deterministic recursion requires \(r\)-dependent concentration (which in this work is handled via Fisher--Kibble/Fisher-type asymptotics and leads to \(O(r^{-1/2})\)-scale fluctuations).

\fi
\paragraph{Step 2: reduce to iterates of \(\omega\).}
With \(z_k=(1-\rho_k)/2\) and \(w_k=z_k^{-1/2}\), we have \(z_{k}=\zeta(z_{k-1})\) and therefore \(w_k=\omega(w_{k-1})\), i.e. \(w_k=\omega^{\circ(k-1)}(w_1)\).

\paragraph{Step 3: asymptotics of \(\omega^{\circ k}\).}
The iterate asymptotic is available in~\cite[Proposition~3]{terjek2025mlpsateoc}.
\end{proof}

\subsection{RF-LR NTK: recursion-driven depth scaling (not the full-MLP closed form)}
For fully trained MLPs at the edge of chaos, the limiting NTK admits an explicit expression in terms of the cosine map and its derivative~\cite[Proposition~4]{terjek2025mlpsateoc}. This relies on training all weight layers. In RF-LR, only the readouts \(A^{(\ell)}\) (and biases \(c^{(\ell)}\)) are trained and the features \(w^{(\ell)}\) are frozen; consequently the RF-LR NTK is described by our recursion (Theorem~\ref{thm:ntk_recursion}) and its closed form (Corollary~\ref{cor:general_L_layer_ntk}, Eq.~\eqref{eq:ntk_explicit_form}) in terms of \(\Sigma^{(\ell)}\) and \(\dot{\Sigma}^{(\ell)}\).

The inverse cosine distance framework is useful here at the level of controlling the depth evolution of correlations (hence of \(\Sigma^{(\ell)}\) and \(\dot{\Sigma}^{(\ell)}\)), rather than by importing the fully trained NTK identity.

\subsection{Inverse cosine distance matrices and spectral bounds}
To formulate dataset-level depth scaling, we work with the deterministic correlation recursion (the mean/infinite-width idealization). Given a dataset \( \{x_1,\ldots,x_n\} \subset \mathbb{R}^{d} \), set
\[
\rho_{1,ij} \;=\; \frac{\langle x_i,x_j\rangle}{\|x_i\|\,\|x_j\|}\in[-1,1],
\qquad
\rho_{k,ij} \;=\; \varrho(\rho_{k-1,ij}),\quad k\ge 2.
\]
Define the layer-\( k \) inverse cosine distance matrix \( W_k \in \mathbb{S}^n \) by
\[
(W_k)_{i,i} = 0,\qquad (W_k)_{i_1,i_2} = \biggl( \frac{1 - \rho_{k,i_1 i_2}}{2} \biggr)^{-1/2},\quad i_1 \neq i_2,
\]
so that \(W_k\) is the inverse cosine distance matrix associated to \(\rho_k\).

\begin{proposition}[Spectral bounds for RF-LR inverse cosine distance matrices]\label{prop:rflr-w-spectral}
Assume EOC initialization and a dataset with no parallel pairs. For each \( k \in [1:L] \) there exists \( \widehat{W}_k \in (1,\infty) \) with \( \widehat{W}_k = \Theta_{W_1}(1) \) such that
\[
\bigl\Vert W_k - \omega^{\circ(k-1)}(\widehat{W}_k)\,(\mathbf{1}_n\mathbf{1}_n^\top - I_n) \bigr\Vert \le O(\Delta_\phi\, n^{-1} k) + O(1).
\]
\end{proposition}
\begin{proof}
By~\cite[Proposition~6]{terjek2025mlpsateoc}, applied to the deterministic recursion \(\rho_{k,ij}=\varrho(\rho_{k-1,ij})\), the inverse cosine distance matrix \(W_k\) is close to the equicorrelated structure \(\omega^{\circ(k-1)}(\widehat{W}_k)(\mathbf{1}\mathbf{1}^\top-I_n)\): convexity and monotonicity of \(\omega\) on \((1,\infty)\) imply the stated matrix-norm bound via standard comparison estimates for the entrywise recursion.
\end{proof}

\subsection{Preliminaries and full statements for Section~\ref{subsec:rflr-terjek-generalization}}\label{appendix:mean-depth-theorem-preliminaries}
We work with the \emph{deterministic proxy} recursion \eqref{eq:rflr-scalar-mean-ntk-rec}, obtained by evaluating the scalar ReLU base/derivative dual kernels along the deterministic ReLU correlation map at the edge of chaos. Starting from an initial cosine similarity \(\rho_1\in(-1,1)\), define the associated correlation recursion by \(\rho_k:=\varrho^{\circ(k-1)}(\rho_1)\) for \(k\ge 2\), where \(\varrho\) is the ReLU EOC cosine map. Then \(\rho_k\) aligns toward \(1\) at the polynomial rate \(1-\rho_k=\Theta(k^{-2})\) as in full-width MLPs at the EOC~\cite{terjek2025mlpsateoc,pmlr-v163-hayou22a}. The explicit proxy recursion, notation, and kernel expansions used in the proof are given below.

\paragraph{Scalar ReLU kernels as functions of cosine similarity.}
For unit-variance ReLU features, write \( \theta=\arccos(\rho) \). The normalized base and derivative kernels are
\[
s(\rho) \;=\; \frac{1}{2\pi}\bigl((\pi-\theta)\cos\theta + \sin\theta\bigr),
\qquad
\dot{s}(\rho) \;=\; \frac{1}{2}-\frac{\theta}{2\pi}.
\]
Near \( \rho=1 \), parameterize by the inverse cosine distance \( w=((1-\rho)/2)^{-1/2} \), so that \( \rho = 1-2w^{-2} \) and \( \theta = 2w^{-1}+O(w^{-3}) \). Then
\begin{equation}\label{eq:relu-kernel-expansions-w}
\dot{s}(\rho) \;=\; \frac{1}{2}-\frac{1}{\pi w}+O(w^{-3}),
\qquad
s(\rho)\;=\;\frac{1}{2}-\frac{1}{w^{2}}+O(w^{-3}).
\end{equation}

\paragraph{Deterministic proxy RF-LR entrywise recursion.}\label{def:deterministic-proxy}
Fix an input pair \((x,x')\) with input cosine similarity \(\rho_1\in(-1,1)\). The \emph{deterministic proxy} replaces the random correlation chain \((\rho_\ell)_{\ell\ge 1}\) (see Appendix~\ref{appendix:random-correlation-chain}) by the deterministic iterates \(\rho_k := \varrho^{\circ(k-1)}(\rho_1)\) and \(w_k := ((1-\rho_k)/2)^{-1/2}\). Define the scalar deterministic proxy RF-LR NTK by
\begin{equation}\label{eq:rflr-scalar-mean-ntk-rec}
\Theta^{(1)}(\rho_1) \;=\; 1 + s(\rho_1),
\qquad
\Theta^{(k)}(\rho_1) \;=\; 1 + \frac{1}{r}\,\Theta^{(k-1)}(\rho_1)\,\dot{s}(\rho_k) + \frac{1}{r}\,s(\rho_k),\quad k\ge 2.
\end{equation}
This is obtained from Theorem~\ref{thm:ntk_recursion} by replacing the random fields \(\Sigma^{(k)},\dot{\Sigma}^{(k)}\) with their scalar ReLU dual functions \(s(\cdot),\dot{s}(\cdot)\) evaluated along the deterministic coefficients \(\rho_k\). The proxy approximates the mean path: in the actual network, \(\rho_\ell\) is random with \(\mathbb{E}[(\rho_\ell - \varrho^{\circ(\ell-1)}(\rho_1))^2] = O(1/r)\), so for large \(r\) the random recursion concentrates around this deterministic path.

\subsection{Proof of Proposition~\ref{prop:condition-number-depth}}\label{appendix:proof-condition-number}

\paragraph{Proof sketch.}
All entries are \(\Theta(1)\). For \(\lambda_{\min}\le O(1/(rL))\): take \(v=e_i-e_j\) for a pair with \(\rho_{ij}<1\); the Rayleigh quotient \(v^\top\mathbf{M}v/2=M_{ii}+M_{jj}-2M_{ij}=\Theta^{(L)}(1)-\Theta^{(L)}(\rho_{ij})=\Theta(1/(rL))\) by Theorem~\ref{thm:rflr-depth-scaling-mean}. For \(\lambda_{\max}=\Theta(1)\): the diagonal is \(\Theta(1)\) and the matrix is positive definite. Hence \(\kappa\ge\Omega(r\cdot L)\).

\begin{proof}
\paragraph{Entries.}
We have \(M_{ij}=\Theta^{(L)}(\rho_{ij})\). The range \(\Theta^{(L)}(\rho)\in[\Theta^{(L)}(-1),\Theta^{(L)}(1)]\) is bounded, and \(\Theta^{(L)}(1)\to\Theta_\star(r)=\Theta(1)\). So all entries are \(\Theta(1)\).

\paragraph{Upper bound on \(\lambda_{\min}\): \(\lambda_{\min}\le O(1/(rL))\).}
By assumption (ii), there exist \(i\neq j\) with \(\rho_{ij}\le\rho_{\max}<1\). Take \(v=e_i-e_j\). Then \(v\perp\mathbf{1}\) and \(\|v\|^2=2\). The Rayleigh quotient is
\[
\frac{v^\top \mathbf{M} v}{\|v\|^2} \;=\; M_{ii}+M_{jj}-2M_{ij} \;=\; \Theta^{(L)}(1)-\Theta^{(L)}(\rho_{ij}).
\]
By Theorem~\ref{thm:rflr-depth-scaling-mean}, \(\Theta^{(L)}(1)-\Theta^{(L)}(\rho)=\Theta(1/(rL))\) for \(\rho<1\). So this quotient is \(O(1/(rL))\), and \(\lambda_{\min}(\mathbf{M}|_{\mathbf{1}^\perp})\le O(1/(rL))\).

\paragraph{\(\lambda_{\max}=\Theta(1)\).}
\emph{Upper bound.} All eigenvalues of \(\mathbf{M}|_{\mathbf{1}^\perp}\) lie in \((0,\Theta^{(L)}(1)]\) because \(\mathbf{M}\) is positive definite and \(M_{ii}=\Theta^{(L)}(1)=\Theta(1)\). Thus \(\lambda_{\max}\le\Theta(1)\).

\medskip\noindent
\emph{Lower bound.} Under assumption (iii) (non-equicorrelated), the off-diagonal entries \(M_{ij}=\Theta^{(L)}(\rho_{ij})\) are not all equal. Pick distinct \(i,j,k\) with \(\rho_{ij}\neq\rho_{ik}\). The restriction of \(\mathbf{M}\) to \(\operatorname{span}\{e_i,e_j,e_k\}\cap\mathbf{1}^\perp\) is a \(2\times 2\) positive definite matrix with diagonal entries \(\Theta(1)\), so its eigenvalues are \(\Theta(1)\). Hence \(\lambda_{\max}(\mathbf{M}|_{\mathbf{1}^\perp})\ge\Omega(1)\).

\paragraph{Condition number.}
Thus \(\kappa=\lambda_{\max}/\lambda_{\min}\ge \Theta(1)/O(1/(rL))=\Omega(r\cdot L)\).
\end{proof}

\subsection{Proof of Corollary~\ref{cor:conditioning-equicorrelated-highdim}}\label{appendix:proof-conditioning-equicorrelated-highdim}
\begin{proof}
\paragraph{Equicorrelated data.}
Assume \(\rho_{ij}=\rho_0\) for all \(i\neq j\) and \(\rho_{ii}=1\). Then
\[
\mathbf{M} \;=\; \Theta^{(L)}(1)I_n + \Theta^{(L)}(\rho_0)(\mathbf{1}\mathbf{1}^\top - I_n) \;=\; \Theta^{(L)}(\rho_0)\mathbf{1}\mathbf{1}^\top + \bigl(\Theta^{(L)}(1)-\Theta^{(L)}(\rho_0)\bigr)I_n.
\]
So \(\mathbf{M}\) has eigenvalue \(\Theta^{(L)}(1)+(n-1)\Theta^{(L)}(\rho_0)\) for the eigenvector \(\mathbf{1}\), and eigenvalue \(\Theta^{(L)}(1)-\Theta^{(L)}(\rho_0)\) with multiplicity \(n-1\) on \(\mathbf{1}^\perp\). By Theorem~\ref{thm:rflr-depth-scaling-mean}, \(\Theta^{(L)}(1)-\Theta^{(L)}(\rho_0)=\Theta(1/(rL))\). On \(\mathbf{1}^\perp\) all eigenvalues therefore equal \(\lambda_\perp=\Theta(1/(rL))\), so \(\kappa_\perp=1\).

\paragraph{High-dimensional random data.}
Let \(x_1,\ldots,x_n\) be i.i.d.\ uniform on \(\mathbb{S}^{d-1}\). For \(i\neq j\), the inner product \(\langle x_i,x_j\rangle\) equals the cosine similarity \(\rho_{ij}\) and satisfies \(\mathbb{P}(|\rho_{ij}|\ge t)\le 2\exp(-c d\,t^2)\) for \(t\ge 0\) (concentration on the sphere~\cite{vershynin2018highdimensionalprobability}). With \(\tau=C/\sqrt{d}\) for a large constant \(C\), we have \(\max_{i\neq j}|\rho_{ij}|=O(1/\sqrt{d})\) with probability \(1-O(n^2)\exp(-\Omega(d))\).

\medskip\noindent
On that event, all off-diagonal \(\rho_{ij}\) lie in an interval of length \(O(1/\sqrt{d})\). The map \(\rho\mapsto \Theta^{(L)}(\rho)\) is Lipschitz on \([-1,1]\) (the recursion has bounded derivative), so \(M_{ij}=\Theta^{(L)}(\rho_{ij})\) for \(i\neq j\) differ by \(O(1/\sqrt{d})\). Thus \(\mathbf{M}\) is an equicorrelated matrix (common off-diagonal \(\Theta^{(L)}(\bar{\rho})\) for some \(\bar{\rho}=O(1/\sqrt{d})\)) plus a matrix \(E\) with \(\|E\|_{\mathrm{F}}\le n^2\cdot O(1/\sqrt{d})=o(1)\) as \(d\to\infty\) with \(n\) fixed.

\medskip\noindent
By Weyl's inequality, each eigenvalue of \(\mathbf{M}|_{\mathbf{1}^\perp}\) differs from the unique eigenvalue \(\lambda_\perp^0=\Theta^{(L)}(1)-\Theta^{(L)}(\bar{\rho})\) of the equicorrelated part by \(O(\|E\|_{\mathrm{op}})=o(1)\). For \(\bar{\rho}=O(1/\sqrt{d})\), the proxy recursion gives \(\Theta^{(L)}(\bar{\rho})\) bounded and \(\Theta^{(L)}(1)=\Theta_\star(r)+O(1/L)=\Theta(1)\), so \(\lambda_\perp^0=\Theta^{(L)}(1)-\Theta^{(L)}(\bar{\rho})\) is \(\Theta(1)\) (for fixed \(r,L\); the \(r,L\)-scaling is \(\Theta(1/(rL))\) by the depth-induced gap). Hence all eigenvalues of \(\mathbf{M}|_{\mathbf{1}^\perp}\) are \(\lambda_\perp^0+o(1)=\Theta(1)(1+o(1))\), and \(\kappa_\perp=1+o(1)\).

\medskip\noindent
On that high-probability event the data are approximately equicorrelated. So the same operator-norm bound on \(\mathbf{1}^\perp\) as in Theorem~\ref{thm:equicorrelated-op-bound} applies to the random Gram matrix \(\hat{K}\): \(\|(\hat{K}-K_{\mathrm{proxy}})|_{\mathbf{1}^\perp}\|_{\mathrm{op}}=O_P(L/r+1/\sqrt{r})\) with high probability (in \(r\) and \(d\)).
\end{proof}

\paragraph{Terjék--González-Sánchez pipeline and RF-LR.}
Terjék and González-Sánchez~\cite{terjek2025mlpsateoc} obtain sharp two-sided eigenvalue and condition-number bounds for full-width MLP NTKs via a three-step pipeline:
\begin{itemize}\setlength{\itemsep}{1pt}
  \item[(i)] Approximate NTK entries by an affine function of the inverse cosine distance \(\omega^{\circ(l-1)}(w)\);
  \item[(ii)] Show the inverse cosine distance matrix is close to an equicorrelated structure;
  \item[(iii)] Transfer spectral bounds via Weyl.
\end{itemize}
For RF-LR we have (ii) in Proposition~\ref{prop:rflr-w-spectral}, but (i) fails: the RF-LR scalar recursion
\[
\Theta^{(k)}(\rho)=1+\frac{1}{r}\,\Theta^{(k-1)}\,\dot{s}(\rho_k)+\frac{1}{r}\,s(\rho_k)
\]
is structurally different from the MLP recurrence, so Terjék's approximation proposition does not apply. We do not have an RF-LR result of the form
\[
\Theta^{(L)}(\rho)\approx A(r)+B(r)\cdot\omega^{\circ(L-1)}\bigl(((1-\rho)/2)^{-1/2}\bigr)/r
\]
with explicit \(A(r),B(r)\) and \(O(1/(rL^2))\) error. Given such an approximation, (iii) would mirror Terjék: the Gram matrix would approximate a rank-one perturbation of a diagonal matrix, yielding sharp \(\lambda_{\min},\lambda_{\max}\) and \(\kappa\) bounds.

\paragraph{Equicorrelated vs general datasets.}
For \emph{equicorrelated} datasets (\(\rho_{ij}=\rho_0\) for \(i\neq j\)), the mean kernel matrix
\[
K=\Theta^{(L)}(1)I+\Theta^{(L)}(\rho_0)(\mathbf{1}\mathbf{1}^\top-I)
\]
has explicit spectrum on \(\mathbf{1}^\perp\), and sharp bounds hold with \(\Theta^{(L)}(1)-\Theta^{(L)}(\rho_0)\asymp 1/(rL)\). For general datasets we are limited to one-sided bounds:
\[
\lambda_{\min}\le O(1/(rL)),\qquad \lambda_{\max}=\Theta(1),\qquad \kappa\ge\Omega(rL).
\]
A lower bound on \(\lambda_{\min}\) and an upper bound on \(\kappa\) would require either an RF-LR analogue of Terjék's NTK--inverse-cosine approximation (e.g.\ by analyzing the recursion with \(\dot{s}(\rho_k)\approx 1/2-1/(\pi w_k)\), \(s(\rho_k)\approx 1/2-1/w_k^2\), and matching asymptotic style) or a perturbation argument for datasets with mild variation in \(\rho_{ij}\).

\subsection{Proof of Theorem~\ref{thm:rflr-depth-scaling-mean}}\label{appendix:proof-rflr-depth-scaling-mean}

\paragraph{Proof sketch.} Step 1 (correlation alignment): Proposition~\ref{prop:rflr-inverse-cosine-prop} gives \(w_k=\omega^{\circ(k-1)}(w_1)\sim c_-k\), hence \(1-\rho_k=2w_k^{-2}=\Theta(k^{-2})\). Step 2 (kernel saturation): Rewrite the recursion as \(\Theta_k=b_k+a_k\Theta_{k-1}\); the limiting coefficients \(a=1/(2r)\), \(b=1+1/(2r)\) yield fixed point \(\Theta_\star(r)=(2r+1)/(2r-1)\). A discrete Grönwall argument shows \(|\Theta_k-\Theta_\star|=O(1/k)\). Step 3 (gap decay): Subtract the diagonal and off-diagonal recursions; the gap \(\Delta_k=\Theta^{\mathrm{diag}}_k-\Theta^{\mathrm{off}}_k\) satisfies \(\Delta_k=a\Delta_{k-1}+\Theta(1/(rk))\), hence \(\Delta_k=\Theta(1/(rk))\).
\begin{proof}
\paragraph{Step 1: correlation alignment.}
By Proposition~\ref{prop:rflr-inverse-cosine-prop}, there exists a constant \(C_w<\infty\) such that for all \(k\ge 1\),
\[
w_k=\omega^{\circ(k-1)}(w_1)
= w_1 + c_0 (k-1) + c_1 \log\!\bigl(c_2 w_1 + k - 1\bigr) + \varepsilon_k,
\qquad |\varepsilon_k|\le C_w,
\]
where \(c_0=\Delta_\phi\,\frac{4}{3\pi}>0\), \(c_1=\Delta_\phi\,\frac{2}{\pi}\), \(c_2=\Delta_\phi^{-1}\frac{3\pi}{4}\).

\medskip\noindent
In particular, \(w_k\ge w_1+c_0(k-1)-C_w\). Choosing \(k_0\) large enough gives \(w_k\ge \frac{c_0}{2}k\) for all \(k\ge k_0\). Similarly, \(w_k\le c_+k\) for some \(c_+<\infty\) and all \(k\ge k_0\). Thus \(c_-k\le w_k\le c_+k\) for \(k\ge k_0\), and \(1-\rho_k=2w_k^{-2}=O(k^{-2})\).

\paragraph{Step 2: kernel saturation.}
Write \(\Theta_k=\Theta^{(k)}(\rho_1)\) and rewrite \eqref{eq:rflr-scalar-mean-ntk-rec} as
\[
\Theta_k=b_k+a_k\Theta_{k-1},\qquad a_k=\frac{1}{r}\dot{s}(\rho_k),\qquad b_k=1+\frac{1}{r}s(\rho_k).
\]
For the limiting constant-coefficient recursion (corresponding to \(\rho_k\to 1\)), we have \(a=\frac{1}{r}\dot{s}(1)=\frac{1}{2r}\), \(b=1+\frac{1}{r}s(1)=1+\frac{1}{2r}\), hence \(\Theta_\star(r)=b/(1-a)=(2r+1)/(2r-1)\).

Set \(e_k=\Theta_k-\Theta_\star(r)\). Then
\[
e_k=a_k e_{k-1} + (b_k-b) + (a_k-a)\Theta_\star(r).
\]
Using \eqref{eq:relu-kernel-expansions-w} and \(w_k\ge c_-k\) for \(k\ge k_0\), there exists \(C<\infty\) such that for all \(k\ge k_0\),
\[
|a_k-a|
=\frac{1}{r}\Bigl|\dot{s}(\rho_k)-\frac12\Bigr|
\le \frac{C}{r}\frac{1}{w_k}
\le \frac{C}{r}\frac{1}{k},
\]
\[
|b_k-b|
=\frac{1}{r}\Bigl|s(\rho_k)-\frac12\Bigr|
\le \frac{C}{r}\frac{1}{w_k^2}
\le \frac{C}{r}\frac{1}{k^2}.
\]

\medskip\noindent
Moreover \(0\le a_k\le a=1/(2r)<1\). Hence for all \(k\ge k_0\),
\[
|e_k|\le a\,|e_{k-1}|+\frac{C'}{k}
\]
for some \(C'<\infty\). A standard induction (discrete Grönwall) yields \(|e_k|\le C_\Theta/k\) for all \(k\ge 1\); in particular \(e_k\to 0\) and \(\Theta_k\to\Theta_\star(r)\).

\medskip\noindent
To make the \(O(1/k)\) rate explicit, fix \(k\ge k_0\) and iterate the one-step inequality:
\[
|e_k|
\le a^{k-k_0}|e_{k_0}| + \sum_{j=k_0+1}^{k} a^{k-j}\frac{C'}{j}.
\]
Since \(j\ge k_0+1\) implies \(1/j\le 1/k_0\) and also \(j\le k\) implies \(1/j\le 1/k\), we can bound
\[
\sum_{j=k_0+1}^{k} a^{k-j}\frac{1}{j}
\le \frac{1}{k}\sum_{t=0}^{k-k_0-1} a^{t}
\le \frac{1}{k}\cdot\frac{1}{1-a}.
\]
Also \(a^{k-k_0}|e_{k_0}|\le |e_{k_0}|\le (k_0|e_{k_0}|)/k\). Therefore for all \(k\ge k_0\),
\[
|e_k|\le \frac{1}{k}\Bigl(k_0|e_{k_0}|+\frac{C'}{1-a}\Bigr).
\]
Absorbing the finitely many values \(k<k_0\) into the constant gives \(|e_k|\le C_\Theta/k\) for all \(k\ge 1\), as claimed.

\medskip
\paragraph{Step 3: diagonal--off-diagonal gap.}
Let \(\Theta^{\mathrm{diag}}_k=\Theta^{(k)}(1)\) and \(\Theta^{\mathrm{off}}_k=\Theta^{(k)}(\rho_1)\), and set \(\Delta_k=\Theta^{\mathrm{diag}}_k-\Theta^{\mathrm{off}}_k\ge 0\). The diagonal recursion is \(\Theta^{\mathrm{diag}}_k=b+a\,\Theta^{\mathrm{diag}}_{k-1}\), while the off-diagonal recursion is \(\Theta^{\mathrm{off}}_k=b_k+a_k\Theta^{\mathrm{off}}_{k-1}\). Subtracting gives the exact identity
\[
\Delta_k
:=a\,\Delta_{k-1} + (a-a_k)\Theta^{\mathrm{off}}_{k-1} + (b-b_k).
\]
For \(k\) large, \(\Theta^{\mathrm{off}}_{k-1}\) is bounded above and below by positive constants (since \(\Theta^{\mathrm{off}}_k\to\Theta_\star(r)\in(1,\infty)\)). Also by \eqref{eq:relu-kernel-expansions-w} and \(w_k\asymp k\),
\[
a-a_k=\frac{1}{r}\Bigl(\frac12-\dot{s}(\rho_k)\Bigr)=\Theta\!\Big(\frac{1}{r\,w_k}\Big)=\Theta\!\Big(\frac{1}{r\,k}\Big),
\qquad
b-b_k=\frac{1}{r}\Bigl(\frac12-s(\rho_k)\Bigr)=O\!\Big(\frac{1}{r\,w_k^2}\Big)=O\!\Big(\frac{1}{r\,k^2}\Big).
\]
Thus there exist \(k_1\) and constants \(0<c<C<\infty\) such that for all \(k\ge k_1\),
\[
a\,\Delta_{k-1}+\frac{c}{r\,k}\le \Delta_k\le a\,\Delta_{k-1}+\frac{C}{r\,k}.
\]
Iterating these comparison recursions yields \(c_\Delta/(rk)\le \Delta_k\le C_\Delta/(rk)\) for all \(k\ge k_1\), completing the proof.
\end{proof}

\subsection{Remark: proxy lower bounds vs positivity (smallest eigenvalue)}\label{appendix:lambda-min-negative}
\paragraph{Notation.}
Let \(\widehat{K}\) denote the empirical NTK Gram matrix at initialization, with entries \(\widehat{K}_{ij}=\langle \nabla_\theta f(x_i),\nabla_\theta f(x_j)\rangle\). Let \(K\) denote the mean (deterministic) kernel Gram matrix with entries \(K_{ij}=\mathbb{E}[\widehat{K}_{ij}]\). The \emph{centered} spectrum refers to the restriction of these matrices to the subspace \(\mathbf{1}^\perp\) of vectors orthogonal to the constant vector \(\mathbf{1}\); we write \(\widehat{K}|_{\mathbf{1}^\perp}\) and \(K|_{\mathbf{1}^\perp}\) for these restrictions.

\begin{remark}[Can the smallest eigenvalue become negative?]\label{rmk:lambda-min-negative-appendix}
No: \(\widehat{K}\) is a Gram matrix of gradients, hence \(\widehat{K}\succeq 0\) deterministically and all its eigenvalues are \(\ge 0\). Restricting to the mean-zero subspace preserves positive semidefiniteness: for any \(v\perp \mathbf{1}\), one has \(v^\top \widehat{K} v\ge 0\), so \(\widehat{K}|_{\mathbf{1}^\perp}\succeq 0\).

What \emph{can} happen is that a deterministic proxy lower bound on the centered eigenvalue scale is smaller than the magnitude of random fluctuations. Weyl's inequality gives
\[
\lambda_{\min}\!\bigl(\widehat{K}|_{\mathbf{1}^\perp}\bigr)\;\ge\;\lambda_{\min}\!\bigl(K|_{\mathbf{1}^\perp}\bigr)\;-\;\|\widehat{K}-K\|_{2},
\]
so if \(\|\widehat{K}-K\|_{2}\) dominates the deterministic scale of \(\lambda_{\min}(K|_{\mathbf{1}^\perp})\), the bound becomes vacuous (it may be negative), but the true eigenvalue still satisfies \(\lambda_{\min}(\widehat{K}|_{\mathbf{1}^\perp})\ge 0\). Thus, the relevant concern is not ``going below zero'' but rather ``being driven close to zero'' when the \(1/(rL)\) signal is swamped by noise.
\end{remark}

\subsection{Remark: infinite-depth limit after centering}\label{appendix:infinite-depth-limit}
\begin{remark}[Infinite-depth limit: existence and degeneracy after centering]\label{rmk:infinite-depth-limit-appendix}
The correlation recursion satisfies \(1-\rho_k=O(k^{-2})\) as in full-width MLPs at the EOC~\cite{terjek2025mlpsateoc,pmlr-v163-hayou22a}, so deep features become increasingly aligned.
For the \emph{mean} RF-LR NTK recursion \eqref{eq:rflr-scalar-mean-ntk-rec}, Theorem~\ref{thm:rflr-depth-scaling-mean} implies that for any fixed \(\rho_1\in(-1,1)\),
\[
\Theta^{(L)}(\rho_1)\to \Theta_\star(r)\qquad\text{and}\qquad
\Theta^{(L)}(1)\to \Theta_\star(r),
\]
so the entrywise infinite-depth limit of the (mean) Gram matrix is a constant kernel: \(K^{(L)}\to \Theta_\star(r)\,\mathbf{1}\mathbf{1}^\top\).
Consequently, after centering (restricting to the orthogonal complement of the constant mode), the relevant part of \(K^{(L)}\) tends to zero and the spectrum collapses.
In particular, a \emph{non-degenerate} deep limit on \(\mathbf{1}^\perp\) can only hold under a joint scaling (e.g. \(r\) growing with \(L\)) or after an explicit rescaling of the centered operator (equivalently, rescaling time/learning rate in kernel gradient descent).
\end{remark}
