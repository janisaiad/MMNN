\section{Network definition and EOC parameterization}

\vspace{1em}
\subsection{RF-LR architecture}
Let \( \bm{h}^{(0)}(\bm{x})=\bm{x}\in\mathbb{R}^{d_0} \). For \( \ell=1,\dots,L \),
\begin{equation}
\label{eq:RF-LR}
\bm{h}^{(\ell)}(\bm{x})
= \frac{\sigma_A}{\sqrt{n_\ell}} \sum_{j=1}^{n_\ell} \bm{A}^{(\ell)}_j\,
\ReLU\!\Big(\bm{w}^{(\ell)\top}_j\,\bm{h}^{(\ell-1)}(\bm{x}) + \beta\, b^{(\ell)}_j\Big) \;+\; \sigma_c\, c.
\end{equation}
\textbf{Training policy:} \( A^{(\ell)} \) and the scalar output bias \( c \) are trained; \( w^{(\ell)} \) and activation biases \( b^{(\ell)} \) are frozen random draws (i.i.d. Gaussian). The \( 1/\sqrt{n_\ell} \) scaling (NTK parameterization) ensures a well-defined infinite-width limit. Low-rank bottlenecks (intermediate dimension \( r\ll n_\ell \)) compress correlations, reduce parameter budget from \( O(n^2) \) to \( O(rn) \), and improve conditioning.

\paragraph{Training parameters, biases, and initialization.}
We use the biased activation and an output bias while keeping all symbols non-bold except the activation \( \ReLU \), which denotes a bold \( \boldsymbol{\sigma} \). The layer map with biases is
\[
h^{(\ell)}(x) \;=\; \frac{\sigma_A}{\sqrt{n_\ell}} \sum_{j=1}^{n_\ell} A^{(\ell)}_j\;\ReLU\!\Big(w^{(\ell)\top}_j\,h^{(\ell-1)}(x) + \beta\, b^{(\ell)}_j\Big) \;+\; \sigma_c\, c, \quad \ell=1,\dots,L.
\]
Initialization is
\[
w^{(\ell)}_j \sim \mathcal{N}\!\big(0, I_{d_{\ell-1}}/d_{\ell-1}\big),\quad
b^{(\ell)}_j \sim \mathcal{N}(0, \beta_\ell^2),\quad
A^{(\ell)}_{ij} \sim \mathcal{N}\!(0, 1),\quad
c \sim \mathcal{N}(0, 1),
\]
We use the edge-of-chaos scaling for the weights so that \( \mathrm{Cov}\!\big(w^{(\ell)}_j\big) = I_{d_{\ell-1}}/d_{\ell-1} \) \cite{hayou2018selectioninitializationactivationfunction}.
The low-rank constraint is enforced as \( \mathrm{rank}\!\big(A^{(\ell)}\big)\le r_\ell \ll n_\ell \).

Remarkably, EOC do not involve any condition on \beta ! This is because it is absorded into the random features manifold.

\paragraph{Why this simplification?}
We include an activation bias \( b^{(\ell)} \) and a scalar output bias \( c \) for completeness, but we avoid introducing extra scaling parameters for them. Output scaling can be absorbed into the learning rate; the mean-field and NTK structures remain qualitatively identical, and the Fourier-space interpretation and stepwise loss phenomenon depend on kernel geometry, not absolute scales.

\paragraph{Activation homogeneity.}
We only assume a positively 1-homogeneous nonlinearity \(\ReLU\) for Kibble/Fisher decoupling:
\[
\ReLU(\alpha u) \,=\, \alpha\,\ReLU(u) \quad \text{for } \alpha \ge 0,\; u \in \mathbb{R}.
\]
This yields the scaling relation \(\ReLU(\alpha\, w^\top h) = \alpha\,\ReLU(w^\top h)\) and, away from zero, \(\dot{\ReLU}(\alpha u)=\dot{\ReLU}(u)\).

\subsection{Parameterization, signal propagation, and kernels}
\[
\text{NTK scaling: } s_\ell \,=\, \frac{1}{\sqrt{n_\ell}}, \qquad
\text{Mean-field scaling: } s_\ell \,=\, \frac{1}{n_\ell}.
\]

Define the per-layer variance
\[
q^{(\ell)} \,=\, \frac{\mathbb{E}\big[\|\bm{h}^{(\ell)}\|^2\big]}{d_\ell}.
\]
For ReLU and centered parameters,
\[
q^{(\ell)} \,=\, \sigma_c^2 \, + \, \frac{\sigma_A^2}{2}\,\Big(q^{(\ell-1)}\,\mathrm{Tr}(\Sigma_w) \, + \, \beta^2\Big),
\]
with \(\Sigma_w = \mathrm{Cov}(w^{(\ell)}_j)\). The edge-of-chaos condition is
\[
\frac{\sigma_A^2}{2}\,\mathrm{Tr}(\Sigma_w) \,=\, 1.
\]
Under \(\Sigma_w = I_{d_{\ell-1}}/d_{\ell-1}\), we obtain \(\sigma_A = \sqrt{2}\) and we set \(\sigma_c^2 = 1\) at EOC.

In the sequential infinite-width limit, the layer-\( \ell \) base kernel is
\begin{equation}
\Sigma^{(\ell)}(x,x')
= \mathbb{E}_{\bm{w}}\Big[\ReLU\!\big(\bm{w}^\top \bm{h}^{(\ell-1)}(x)\big)\,\ReLU\!\big(\bm{w}^\top \bm{h}^{(\ell-1)}(x')\big)\Big],
\end{equation}
and the derivative kernel is
\begin{equation}
\dot{\Sigma}^{(\ell)}(x,x')
= \mathbb{E}_{\bm{w}}\Big[\dot{\ReLU}\!\big(\bm{w}^\top \bm{h}^{(\ell-1)}(x)\big)\,\dot{\ReLU}\!\big(\bm{w}^\top \bm{h}^{(\ell-1)}(x')\big)\;\bm{w}^\top\bm{w}\Big].
\end{equation}
For \( \ell=1 \) and centered Gaussian \( \bm{w} \), \( \Sigma^{(1)} \) is rotation-invariant with standard closed forms in terms of input cosine similarity \( \rho = \langle x,x'\rangle/(\|x\|\|x'\|) \):
\[
\Sigma^{(1)}(x,x') = \frac{\|x\|\|x'\|}{2\pi}\big((\pi-\theta)\cos\theta + \sin\theta\big), \quad \theta = \arccos(\rho).
\]

\subsection{NTK recursion}
\begin{theorem}[Infinite-width NTK recursion]
\label{thm:ntk_recursion}
Under NTK parameterization \( 1/\sqrt{n_\ell} \) and a sequential limit \( n_1\to\infty,\dots,n_L\to\infty \) with \( \bm{w}^{(\ell)} \sim \mathcal{N}(0, \mathbf{I}_{d_{\ell-1}}) \) i.i.d. frozen after initialization, the layer-\( \ell \) NTK satisfies
\begin{align}
\Kop^{(0)}(x,x') \;&=\; 0, \\
\Kop^{(\ell)}(x,x') \;&=\; \Kop^{(\ell-1)}(x,x')\cdot\dot{\Sigma}^{(\ell)}(x,x')\;+\;\Sigma^{(\ell)}(x,x'),\quad \ell\ge1,
\end{align}
where
\begin{align}
\Sigma^{(\ell)}(x,x') &= \mathbb{E}_{\bm{w}}\Big[\ReLU\!\big(\bm{w}^\top \bm{h}^{(\ell-1)}(x)\big)\,\ReLU\!\big(\bm{w}^\top \bm{h}^{(\ell-1)}(x')\big)\Big], \\
\dot{\Sigma}^{(\ell)}(x,x') &= \mathbb{E}_{\bm{w}}\Big[\dot{\ReLU}\!\big(\bm{w}^\top \bm{h}^{(\ell-1)}(x)\big)\,\dot{\ReLU}\!\big(\bm{w}^\top \bm{h}^{(\ell-1)}(x')\big)\;\bm{w}^\top\bm{w}\Big].
\end{align}
For \( \ell=1 \), \( \Sigma^{(1)} \) is deterministic; for \( \ell>1 \), \( \Sigma^{(\ell)} \) and \( \dot{\Sigma}^{(\ell)} \) are random fields whose fluctuations shrink with width and with the rank \( r \) of bottlenecks.

Furthermore, the gradient flow dynamics in the NTK regime satisfy
\[
\frac{d}{dt}f_t(x) = -\int \Kop^{(L)}(x,x') \big(f_t(x') - y(x')\big) d\mu(x'),
\]
where \( \mu \) is the data distribution and \( y \) is the target function.
\end{theorem}

Proof: See Appendix A.1.

\paragraph{Interpretation.} The additive term \( \Sigma^{(\ell)} \) encodes fresh basis contribution at layer \( \ell \), while the multiplicative coupling \( \Kop^{(\ell-1)}\cdot\dot{\Sigma}^{(\ell)} \) transports geometry through layers via the derivative kernel.
