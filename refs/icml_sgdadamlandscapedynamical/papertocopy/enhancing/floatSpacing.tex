\makeatletter
\def\infigenv#1{\ifthenelse{\equal{\@currenvir}{figure}}{#1}{}}
\def\intabenv#1{\ifthenelse{\equal{\@currenvir}{table}}{#1}{}}
\makeatother


\let\mycaption\caption
\def\caption#1{
	\infigenv{\vskip -6.6pt} % skip before cap
	\mycaption{#1}
	\intabenv{\vskip  5pt} % skip after cap
	}
